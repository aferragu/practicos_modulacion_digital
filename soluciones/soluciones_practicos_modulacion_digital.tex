\section{Codificación}

\begin{ejercicio}[Codificación de Huffman de una fuente no diádica]
Fuente con alfabeto $\{A,B,C,D\}$ y probabilidades $0.4$, $0.3$, $0.2$ y $0.1$. Entropía, código de Huffman, largo medio y eficiencia, codificación de $ACABADAC$ y conveniencia de codificar en bloques de $N=2$.
\end{ejercicio}

\begin{solucion}
\begin{enumerate}
\item $H(X)=-\sum_i p_i\log_2 p_i = 0.4(1.322)+0.3(1.737)+0.2(2.322)+0.1(3.322)\approx 1.846$ bits/símbolo.
\item Se combinan primero $D$ y $C$ ($0.1+0.2=0.3$), luego ese nodo con $B$ ($0.3+0.3=0.6$) y por último con $A$. Una asignación posible es
\begin{center}
\begin{tabular}{c|c|c|c}
símbolo & $p_i$ & palabra & $\ell_i$\\ \hline
$A$ & $0.4$ & \texttt{0} & $1$\\
$B$ & $0.3$ & \texttt{10} & $2$\\
$C$ & $0.2$ & \texttt{110} & $3$\\
$D$ & $0.1$ & \texttt{111} & $3$
\end{tabular}
\end{center}
(los empates pueden resolverse de otra forma, con el mismo largo medio).
\item $\bar L = 0.4+0.6+0.6+0.3=1.9$ bits/símbolo, y $\eta=1.846/1.9\approx 97.2\%$.
\item $A\,C\,A\,B\,A\,D\,A\,C\mapsto$ \texttt{0\,110\,0\,10\,0\,111\,0\,110}, de $15$ bits, frente a $16$ bits con un código de largo fijo. El código es prefijo, por lo que la cadena se decodifica sin ambigüedad.
\item Con bloques de $N=2$ ($16$ símbolos dobles) el código de Huffman da $\bar L_2/2\approx 1.865$ bits/símbolo, es decir $\eta\approx 99.0\%$. La mejora es de menos de $2\%$ a cambio de una tabla de $16$ palabras: como la eficiencia ya era alta, la ganancia es marginal.
\end{enumerate}
\end{solucion}

\begin{ejercicio}[Capacidad de Shannon y WiFi]
Canal de $B=20\ \unit{\mega\hertz}$ con $\mathrm{SNR}=22\ \unit{\decibel}$, y tasa real de $65\ \unit{\mega\bps}$.
\end{ejercicio}

\begin{solucion}
$\mathrm{SNR}=10^{2.2}\approx 158.5$, por lo que $C=B\log_2(1+\mathrm{SNR})=20\times10^6\times 7.32\approx 146\ \unit{\mega\bps}$, es decir $7.32\ \unit{\bps/\hertz}$. La tasa real da $65/20=3.25\ \unit{\bps/\hertz}$, un $44\%$ de la cota de Shannon. En el capítulo de OFDM se ve de dónde sale ese valor: $52$ subportadoras de datos, $6$ bits por símbolo $64$-QAM, código de tasa $5/6$ y un símbolo OFDM cada $4\ \unit{\micro\second}$: $52\times 6\times\frac56/4\ \unit{\micro\second}=65\ \unit{\mega\bps}$.
\end{solucion}

\begin{ejercicio}[Bit de paridad]
Bloques de $N$ bits de datos más un bit de paridad, canal con probabilidad de error de bit $p=10^{-4}$.
\end{ejercicio}

\begin{solucion}
\begin{enumerate}
\item El bloque transmitido tiene $N+1$ bits. El error no se detecta si hay un número par, no nulo, de bits invertidos:
\begin{equation*}
P_{\text{no detectado}}=\sum_{\substack{k=2\\ k \text{ par}}}^{N+1}\binom{N+1}{k}p^k(1-p)^{N+1-k}.
\end{equation*}
\item Si $p\ll1$ (y $Np\ll 1$), cada término es del orden de $(Np)^k$, por lo que domina el de $k=2$, en el que además $(1-p)^{N-1}\approx 1$. Queda $P_{\text{no detectado}}\approx\binom{N+1}{2}p^2$.
\item
\begin{center}
\begin{tabular}{c|ccccc}
$N$ & $8$ & $16$ & $32$ & $64$ & $128$\\ \hline
tasa $N/(N+1)$ & $0.889$ & $0.941$ & $0.970$ & $0.985$ & $0.992$\\
$P_{\text{no detectado}}$ & $3.6\times10^{-7}$ & $1.4\times10^{-6}$ & $5.3\times10^{-6}$ & $2.1\times10^{-5}$ & $8.3\times10^{-5}$
\end{tabular}
\end{center}
Al aumentar $N$ el overhead baja como $1/N$, pero la probabilidad de error no detectado crece como $N^2$.
\item $\frac{(N+1)N}{2}10^{-8}<10^{-6}\iff N(N+1)<200$, de donde $N_{\max}=13$.
\end{enumerate}
\end{solucion}

\begin{ejercicio}[Código de Hamming(7,4)]
Matrices $G=(I_4\mid P)$ y $H=\begin{pmatrix}P\\ I_3\end{pmatrix}$, codificación $c=mG$ y síndrome $s=rH$.
\end{ejercicio}

\begin{solucion}
\begin{enumerate}
\item $c=mG=(mI_4\mid mP)=(m,mP)$: los primeros $4$ bits son el mensaje y $d=mP$ son los $3$ bits de paridad.
\item $GH=I_4P+PI_3=P+P$, que es nula en aritmética módulo $2$. Entonces $cH=mGH=0$ para todo mensaje.
\item $s=(c+e_i)H=cH+e_iH=e_iH$, que es la fila $i$ de $H$. Como las $7$ filas son distintas y no nulas, el síndrome identifica sin ambigüedad la posición $i$, y basta invertir ese bit.
\item $s=e_iH+e_jH$ es la suma de dos filas distintas, que nunca es nula. El error se detecta.
\item Pero esa suma es otra fila de $H$, por lo que el receptor ``corrige'' un tercer bit y queda con tres bits erróneos. Por ejemplo, con $m=1000$, $c=1000110$, e inversión de los bits $1$ y $2$, $r=0100110$ y $s=110+101=011$, la fila $3$. El receptor invierte el bit $3$ y decodifica $m=0110$. El código corrige un error \emph{o} detecta dos, pero no ambas cosas a la vez.
\end{enumerate}
\end{solucion}

\section{Procesos estocásticos}

\begin{ejercicio}[Media, autocorrelación y densidad espectral de una fuente iid]
$a_k\in\{-1,0,4\}$ independientes, con probabilidades $0.5$, $0.3$ y $0.2$.
\end{ejercicio}

\begin{solucion}
\begin{enumerate}
\item $\mu=-0.5+0.8=0.3$, $P=\E{a_k^2}=0.5+3.2=3.7$, $\sigma^2=3.7-0.09=3.61$.
\item $R_a[0]=P=3.7$ y, por independencia, $R_a[l]=\mu^2=0.09$ para $l\neq0$. Es decir $R_a[l]=3.61\,\delta[l]+0.09$.
\item $S_a(\phi)=\sum_l R_a[l]\e^{-j2\pi l\phi}=3.61+0.09\sum_{k}\delta(\phi-k)$: un piso constante (la parte aleatoria) más un impulso en cada entero (la media), periódico de período $1$.
\end{enumerate}
\end{solucion}

\begin{ejercicio}[Variables en cuadratura]
$x(t)=X\cos(2\pi f_0t)+Y\sin(2\pi f_0t)$, con $X,Y$ independientes, de media nula y varianza $\sigma^2$.
\end{ejercicio}

\begin{solucion}
$\mu(t)=0$. Como $\E{XY}=0$,
\begin{equation*}
R(t,s)=\sigma^2\left[\cos(2\pi f_0t)\cos(2\pi f_0s)+\sin(2\pi f_0t)\sin(2\pi f_0s)\right]=\sigma^2\cos(2\pi f_0(t-s)).
\end{equation*} Es WSS, con $P=\sigma^2$. Si $\sigma_X^2\neq\sigma_Y^2$,
\begin{equation*}
R(t,s)=\frac{\sigma_X^2+\sigma_Y^2}{2}\cos(2\pi f_0\tau)+\frac{\sigma_X^2-\sigma_Y^2}{2}\cos(2\pi f_0(t+s)),
\end{equation*}
que depende de $t+s$: el proceso deja de ser WSS (su potencia instantánea oscila con el tiempo).
\end{solucion}

\begin{ejercicio}[Suma de dos sinusoides de fase aleatoria]
$x(t)=A_1\cos(2\pi f_1t+\theta_1)+A_2\cos(2\pi f_2t+\theta_2)$.
\end{ejercicio}

\begin{solucion}
$\mu(t)=0$, porque cada sinusoide de fase uniforme tiene media nula. Al desarrollar $\E{x(t)x(s)}$ aparecen dos términos ``diagonales'', que dan $\frac{A_i^2}{2}\cos(2\pi f_i\tau)$ como en el teórico, y dos cruzados, que se anulan porque $\theta_1$ y $\theta_2$ son independientes y cada factor tiene media nula. Así $R_x(\tau)=\frac{A_1^2}{2}\cos(2\pi f_1\tau)+\frac{A_2^2}{2}\cos(2\pi f_2\tau)$, WSS, con $P=\frac{A_1^2+A_2^2}{2}$.

Si $\theta_1=\theta_2=\theta$, los términos cruzados ya no se anulan: 
\begin{equation*}
\E{\cos(2\pi f_1t+\theta)\cos(2\pi f_2s+\theta)}=\frac12\cos(2\pi f_1t-2\pi f_2s)
\end{equation*}
 (el término en $2\theta$ sí promedia cero). Como $f_1\neq f_2$, esto depende de $t$ y $s$ por separado, y el proceso no es WSS.
\end{solucion}

\begin{ejercicio}[Filtrado de un proceso WSS]
Proceso WSS de media $\mu_x$ a través de un sistema LTI con $\hat H(0)=G$.
\end{ejercicio}

\begin{solucion}
$\mu_y=\E{\int h(u)x(t-u)du}=\mu_x\int h(u)du=\hat H(0)\mu_x=G\mu_x$. Si $\mu_x=0$, entonces $\mu_y=0$ para cualquier sistema LTI, ya que la salida es una combinación lineal (integral) de valores de la entrada, todos de media nula. Para el pasabajos ideal, $S_y(f)=|\hat H(f)|^2S_x(f)$ y $P_y=G^2\int_{-B}^{B}S_x(f)\,df$.
\end{solucion}

\begin{ejercicio}[Ruido blanco filtrado: probabilidad de exceder un umbral]
$\eta_0/2=2\times10^{-12}\ \unit{\square\volt/\hertz}$, pasabajos ideal de ancho $B=1\ \unit{\mega\hertz}$.
\end{ejercicio}

\begin{solucion}
$N=\frac{\eta_0}{2}\cdot 2B=4\times10^{-6}\ \unit{\square\volt}$, de donde $\sigma=2\ \unit{\milli\volt}$. Como $n(t)$ es gaussiano de media nula, 
\begin{equation*}
\p{|n(t)|>6\ \unit{\milli\volt}}=2Q(6/2)=2Q(3)\approx 2.7\times10^{-3}.
\end{equation*}
\end{solucion}

\section{Modulación en banda base}

\begin{ejercicio}
Un sistema de comunicación digital envía una señal PAM digital NRZ cuaternaria, con símbolos independientes $a_k=\{0,A,2A,3A\}$. Las probabilidades de ocurrencia de cada símbolo son $2/5$, $2/5$, $1/10$ y $1/10$, respectivamente. Calcule y bosqueje la densidad espectral de potencia de la señal trasmitida.
\end{ejercicio}

\begin{solucion}

Calculamos primero $\mu$ y $\sigma^2$ de los símbolos:
\begin{align*}
        \mu &= 0 (2/5) + A (2/5) + 2A (1/10) + 3A (1/10) = 0.9 A \Rightarrow \mu^2 = 0.81 A^2\\
        E[a_k^2] &= 0 (2/5) + A^2 (2/5) + 4A^2 (1/10) + 9A^2 (1/10) = 1.7 A^2\\
        \sigma^2 &= E[a_k^2] - \mu^2 = 0.89 A^2
\end{align*}

Recordamos ahora la fórmula de la densidad espectral de potencia:
\begin{equation*}
        S_x(f) = \sigma^2 f_s |\hat{P}(f)|^2 + \mu^2 f_s^2 \sum_k |P(kf_s)|^2 \delta(f-k f_s)
\end{equation*}

Como $p(t) = \mathbf{1}_{[0,T_s)}(t)$ se tiene que:
\begin{equation*}
        |\hat{P}(f)| = T_s \sinc (\pi T_s f) \; \Rightarrow \; |\hat{P}(f)|^2 = T_s^2 \sinc^2 (\pi T_s f)
\end{equation*}
Observemos que el $\sinc$ se anula en $k f_s$, $k\neq 0$, por lo que solo hay término de continua en la parte discreta. El espectro final es:
\begin{equation*}
        S_x(f) = 0.89 A^2 T_s \sinc^2(\pi T_s f) + 0.81 A^2 \delta(f)
\end{equation*}

\end{center}

\end{solucion}

\begin{ejercicio}
        Sea una señal PAM digital polar binaria $x(t)$ que utiliza pulsos rectangulares $p(t)$ de ancho $T_s$, con formato RZ (retorno a cero en $T_z<T_s$).
        \begin{enumerate}
        \item Asumiendo que los símbolos $a_k=\{A,-A\}$ son independientes y equiprobables, bosqueje la densidad espectral de potencia y calcula la potencia de la señal $\langle x^2(t)\rangle$.
        \item Supongamos ahora que la trasmisión es en formato NRZ y $P(A)=\alpha$ y $P(-A)=1-\alpha$, es decir, los símbolos no son equiprobables. Demuestre que $\langle x^2(t)\rangle$ no depende de $\alpha$.
        \end{enumerate}                
\end{ejercicio}
        
\begin{solucion}

        \begin{enumerate}
        \item Calculamos primero $\mu$ y $\sigma^2$ de los símbolos:
        \begin{align*}
                \mu &= -A (1/2) + A (1/2) = 0 \\
                \sigma^2 &= E[a_k^2] = A^2 (1/2) + A^2(1/2) = A^2
        \end{align*}
        
        Calculamos ahora la transformada del pulso de tiempo $T_z<T_s$:
        \begin{equation*}
                |\hat{P}(f)| = T_z \sinc (\pi T_z f) \; \Rightarrow \; |\hat{P}(f)|^2 = T_z^2 \sinc^2 (\pi T_z f)
        \end{equation*}
        Donde se usó que el pulso es de ancho $T_z$.

        Como $\mu=0$ la fórmula de densidad espectral de potencia queda:
        \begin{equation*}
                S_x(f) = \sigma^2 f_s |\hat{P}(f)|^2
        \end{equation*}
        
        El espectro final es:
        \begin{equation*}
                S_x(f) = A^2 f_s T_z^2 \sinc^2(\pi T_z f)
        \end{equation*}
        Notar que $f_s$ y $f_z = 1/T_z$ ahora son distintos!! Los pulsos son más rápidos que el tiempo de símbolo ($T_z<T_s \Rightarrow f_z > f_s$) por lo que ocupa más ancho de banda.

        Ejemplo: $T_z = T_s/2$, medio símbolo, entonces $f_z = 2f_s$.
        \begin{center}
        \begin{tikzpicture}
                \pgfmathdeclarefunction{sinc}{1}{%
                \pgfmathparse{(#1==0 ? 1: sin(#1 r))/(#1==0 ? 1: #1)}%
                }
                \begin{axis}[x=2cm, y=2cm, axis lines=middle,  
                        xlabel=$f$,
                        x label style={anchor=north},
                        y label style={anchor=east},
                        xmin=-3,xmax=3,ymin=0,ymax=1.5,
                        xtick={-2,-1,0,1,2}, ytick=\empty,
                        xticklabels={$-2f_s$,$-f_s$, $0$, $f_s$, $2f_s$},
                        clip=false,
                        smooth
                    ]
                \addplot+[thick,mark=none,teal, domain = (-3:3), samples=100] {sinc(pi*.5*x)^2} node[midway,above right] {$\frac{A^2 T_s}{4}$};
        
                \end{axis}
        \end{tikzpicture}
        \end{center}
        
        Para hallar la potencia total hay que integrar $S_x(f)$ de la siguiente forma (usando Parseval):
        \begin{equation*}
                S = \int_{-\infty}^\infty S_x(f)df = \sigma^2 f_s \int_{-\infty}^\infty |\hat{P}(f)|^2 df = \sigma^2 f_s \int_{-\infty}^\infty p^2(t)dt,
        \end{equation*}
        De donde:
        \begin{equation*}
                S = \sigma^2 f_s T_z = A^2 f_s T_z.
        \end{equation*}

        
        \item Si ahora $P(A) = \alpha$, los símbolos no son equiprobables, recalculamos $\mu$ y $\sigma^2$. Hacemos la cuenta para un pulso de ancho $T_z$ genérico, y el caso NRZ pedido es $T_z=T_s$. (En ese caso es inmediato: $\sinc(k\pi)=0$ para $k\neq 0$, solo queda la delta en continua, y $S=4\alpha(1-\alpha)A^2+(2\alpha-1)^2A^2=A^2$.)
        
        \begin{align*}
                \mu &= -A (1-\alpha) + A (\alpha) = A(2\alpha-1) \\
                E[a_k^2] &= A^2 (1-\alpha) + A^2(\alpha) = A^2 \\
                \sigma^2 &= A^2 - A^2(2\alpha-1)^2 = 4\alpha(1-\alpha)A^2.
        \end{align*}

        El espectro ahora presenta parte discreta, ya que $\mu\neq 0$ y $\hat{P}(f)$ no necesariamente se anula en los múltiplos de $f_s$ (se anula en múltiplos de $f_z$).

        El espectro completo es:
        \begin{equation*}
                S_x(f) = 4\alpha(1-\alpha)A^2 f_s T_z^2 \sinc^2(\pi T_z f) + A^2(2\alpha-1)^2 \sum_{k} T_z^2 \sinc^2(\pi T_z k f_s) \delta(f-kf_s)
        \end{equation*}
        Integrando en todas las frecuencias y usando que la parte continua no cambia su integral:
        \begin{equation*}
                S = \int_{-\infty}^\infty S_x(f)df = 4\alpha(1-\alpha)A^2 f_s T_z + A^2(2\alpha-1)^2 f_s  \sum_k \left[\frac{T_z}{T_s} \sinc(k\pi T_z f_s)\right]^2.
        \end{equation*}

        Notar que el último término tiene forma de Parseval de una serie de Fourier. De hecho, los coeficientes $c_k = \frac{T_z}{T_s} \sinc(k\pi T_z f_s)$ corresponden a una onda cuadrada de ancho $T_z$, período $T_s$ y altura $1$. Usando Parseval, la serie de coeficientes es entonces igual a la integral en un período de dicho pulso al cuadrado, es decir $T_z$.

        Sustituyendo queda:
        \begin{equation*}
                S = 4\alpha(1-\alpha)A^2 f_s T_z + A^2(2\alpha-1)^2 f_s T_z = A^2 f_s T_z
        \end{equation*}
        que no depende de $\alpha$ (en particular es el mismo valor que antes, $\alpha=1/2$).

        La interpretación del resultado es que, independientemente de $\alpha$, $x^2(t)$ es una onda cuadrada de alto $A^2$, ancho $T_z$ y período $T_s$, por lo que la potencia media transmitida es $A^2T_z f_s$, el valor medio de dicha onda.
        \end{enumerate}
\end{solucion}
        


\begin{ejercicio}
Una computadora genera palabras de 16 bits a una velocidad de 20.000 palabras por segundo. Determinar el ancho de banda mínimo necesario para trasmitirlas usando un sistema de señalización binario. Calcule el número de símbolos mínimo de un sistema M-ario si el ancho de banda de trasmisión disponible es de 60\unit{\kHz}.
\end{ejercicio}

\begin{solucion}
        Velocidad de transmisión de información requerida:
        \begin{equation*}
                \mathrm{vti} = 16 \times 20000 = 320 \, \textrm{kbps}
        \end{equation*}
        El sistema es binario, por lo que el mínimo ancho de banda corresponde a usar pulsos de Nyquist con $\mathrm{vti} = f_s = 2B$ de donde el mínimo es $160 \unit{\kilo\hertz}$.

        Si en cambio, el ancho de banda disponible es $60 \unit{\kilo\hertz}$, la frecuencia de símbolos máxima es $f_s=2B=120 \textrm{kbaud/s}$. Necesito hallar $M$ tal que:
        \begin{equation*}
                 \mathrm{vti} = \log_2(M) f_s \Rightarrow M \geqslant 2^{\frac{320}{120}} \approx 6.35 \mapsto M=7 \textrm{ símbolos}
        \end{equation*}
        En la práctica usaríamos $8=2^3$ y un ancho de banda algo menor ($\approx 53 \unit{\kilo\hertz}$).
\end{solucion}

\begin{ejercicio}
        Un dispositivo debe transmitir información a una tasa de 56kbps utilizando señalización binaria y pulsos de Nyquist con exceso de ancho de banda $\alpha$.
        \begin{enumerate}
        \item Calcule el ancho de banda necesario para la transmisión para $\alpha\in\left\{0,\frac{1}{4},\frac{1}{2},1\right\}$.
        \item Suponga ahora que el canal tiene un ancho de banda de 30kHz y se utiliza señalización $M$-aria. Calcule el mínimo valor de $M$ necesario en cada caso para los valores de $\alpha$ utilizados en la parte anterior.
        \end{enumerate}
\end{ejercicio}

\begin{solucion}

        \begin{enumerate}
        \item $\textrm{vti} = 56 \textrm{ kbps}$, señalización binaria por lo que $f_s=\textrm{vti} = 56 \unit{\kilo\hertz}$. Para pulsos de Nyquist con exceso $\alpha$ vale:
        \begin{equation*}
                \frac{f_s}{2}(1+\alpha) = B
        \end{equation*}
        De donde los anchos de banda son $28 \unit{\kilo\hertz}$, $35 \unit{\kilo\hertz}$, $42 \unit{\kilo\hertz}$ y $56 \unit{\kilo\hertz}$ respectivamente.

        \item Si el canal es de $30 \unit{\kilo\hertz}$, las posibles frecuencias de símbolo son $f_s = \frac{2B}{1+\alpha}$, es decir $f_s = 60 \textrm{ kbaud}$, $48 \textrm{ kbaud}$, $40 \textrm{ kbaud}$ y $30 \textrm{ kbaud}$ respectivamente. Se necesita $\log_2 M\geqslant \textrm{vti}/f_s$, es decir $M\geqslant 2^{\textrm{vti}/f_s}$, que da $1.91$, $2.25$, $2.64$ y $3.65$ respectivamente. Tomando el menor entero que cumple la desigualdad: $M=2$ para $\alpha=0$, $M=3$ para $\alpha=\frac14$ y $\alpha=\frac12$, y $M=4$ para $\alpha=1$.
        \end{enumerate}
\end{solucion}

\section{Error en detección sincrónica en presencia de ruido}

\begin{ejercicio}
Se considera un sistema de trasmisión digital con señalización binaria unipolar $a_k=\{0,A\}$, pulsos rectangulares NRZ, y donde el ruido en el detector es Gaussiano con media nula y varianza $\sigma^2$. Si la probabilidad de error de símbolo es $P_e=1\times 10^{-5}$, calcular la relación $\left(\frac{A}{\sigma}\right)$. Si en cambio se usa señalización polar con igual relación señal a ruido (es decir, con la misma potencia media de señal y el mismo ruido), ¿cuál es la probabilidad de error de símbolo? ¿Y si se mantiene la misma separación $A$ entre los dos niveles?
\end{ejercicio}

\begin{solucion}
Con señalización unipolar el umbral es $A/2$ y necesitamos $Q(A/(2\sigma)) = 10^{-5}$. De la tabla, $A/(2\sigma) = 4.265$, o bien $A/\sigma = 8.53$.

La potencia media de la señal unipolar es $A^2/2$. Una señal polar $\pm A_p$ con la misma potencia tiene $A_p^2=A^2/2$, es decir $A_p=A/\sqrt2$, y con umbral $0$ la probabilidad de error es $Q(A_p/\sigma)=Q(8.53/\sqrt2)=Q(6.03)\approx 8\times10^{-10}$.

Si en cambio se mantiene la separación $A$ entre niveles, los símbolos son $\pm A/2$, el margen de ruido es el mismo, $A/2$, y la probabilidad de error es la misma, $10^{-5}$, pero con la mitad de la potencia. (Si se tomara $\pm A$, con el doble de separación y cuatro veces la potencia del caso anterior, se obtendría $Q(8.53)\approx 7\times10^{-18}$.)
\end{solucion}


\begin{ejercicio}
Se considera un sistema de transmisión con señalización binaria polar, donde existe interferencia intersimbólica (ISI). Modelamos esta interferencia como una tensión $\pm \alpha A$ que se añade al valor de señal en el instante de muestreo, con $0 < \alpha < 1$. Suponiendo que tanto los símbolos $\pm A$ como los valores de interferencia $\pm \alpha A$ son equiprobables e independientes entre sí, determinar la probabilidad de error de símbolo. En el caso $A=8\sigma$ y $\alpha=0.1$ comparar las probabilidades de error con y sin ISI.
\end{ejercicio}

\begin{solucion}
        Notemos que los símbolos posibles a recibir van a ser $\{-(1+\alpha)A, -(1-\alpha)A, (1-\alpha)A, (1+\alpha)A\}$, todos con igual probabilidad (todas las combinaciones de símbolo e ISI modelada). Al detectar tendremos un ruido Gaussiano de potencia $\sigma^2$. Colocamos el umbral el $0$ por simetría y la probabilidad de error queda:
        \begin{equation*}
                P_e = \frac{Q((1-\alpha)(A/\sigma))+Q((1+\alpha)(A/\sigma))}{2}
        \end{equation*}

        Para $A/\sigma=8$ y $\alpha=0.1$ queda:
        \begin{equation*}
                P_e = \frac{Q(7.2)+Q(8.8)}{2} \approx 1.5\times 10^{-13}.
        \end{equation*}
        Para $\alpha=0$ (sin ISI), queda $Q(8)= 6\times 10^{-16}$. La ISI aumenta mucho la probabilidad de error, de hecho, la mitad del tiempo estamos trabajando con un un $A$ efectivo de $(1-\alpha)A$ y en general $Q((1+\alpha)(A/\sigma)) \ll Q((1-\alpha)(A/\sigma))$ por lo que $P_e \approx Q((1-\alpha)(A/\sigma))/2$.
\end{solucion}

        \noindent\begin{minipage}{0.4\textwidth}
        \begin{equation*}
        f_x(x\mid A)= \frac{1}{\sqrt{2\sigma^2}} e^{-\sqrt{2}|x-A|/\sigma}
        \end{equation*}
        \end{minipage}%
        \begin{minipage}{0.6\textwidth}
        \begin{center}
        \begin{tikzpicture}
                \begin{axis}[   x=1cm, y=1.5cm, axis lines=middle,

\begin{equation*}
        P(x<0\mid A) = \int_{-\infty}^0 \frac{1}{\sqrt{2\sigma^2}} e^{-\sqrt{2}|x-A|/\sigma} dx = \frac{1}{\sqrt{2\sigma^2}} \int_{-\infty}^0 e^{-\frac{\sqrt{2}}{\sigma} (A-x)}dx = \frac{1}{2} e^{-\sqrt{2}\frac{A}{\sigma}}.
\end{equation*} 
Por simetría $P(x>0\mid -A)$ es igual, y por lo tanto la probabilidad de error es:

        P_e = \frac{1}{2}P(x<0\mid A)+ \frac{1}{2}P(x>0\mid -A) = \frac{1}{2} e^{-\sqrt{2}\frac{A}{\sigma}}.

\begin{ejercicio}
Se recibe una señal $\sum_k a_k p(t-kT_s)$, con símbolos unipolares binarios $a_k
\in \{0,A\}$, y pulsos triangulares $ p(t)=1 - \frac{|t|}{T_s}$ para $|t|\leqslant T_s$ (de duración $2T_s$), contaminada con ruido blanco de densidad espectral $\frac{\eta_0}{2}$.
\begin{enumerate}
\item Especificar el detector óptimo (filtro acoplado) para esta forma de pulso.
\item Calcular la probabilidad de error de símbolo en función de los parámetros $A$, $T_s$ y $\eta_0$, suponiendo que no hay interferencia intersimbólica.
\item Verificar que el pulso $p(t)$ cumple la condición de ISI nula, pero que el pulso a la salida del filtro acoplado no la cumple. Calcular la ISI en el instante de muestreo y discutir si la fórmula de la parte anterior es válida. Comparar con el pulso de duración $T_s$ del Ejercicio~\ref{ej:triangular corto}.
\end{enumerate}
\end{ejercicio}

\begin{solucion}
        Al ser un pulso triangular, el filtro acoplado correspondiente (causal) es otro triángulo $h(t) = 1-\frac{|t-T_s|}{T_s}$ entre $0$ y $2T_s$, y muestrear con $\tau=T_s$ de retardo. Podemos además calcular $||p||^2$ integrando en el tiempo para obtener $||p||^2 = \frac{2}{3}T_s$ y ajustar la ganancia $\kappa=1/||p||^2 = \frac{3}{2T_s}$ para recuperar el símbolo exacto a la salida.

                \begin{tikzpicture}
                        \begin{axis}[   x=3cm, y=1.5cm, axis lines=middle,

                                ylabel=$p(t)$,

                                clip=false,

                        \end{axis}
                \end{tikzpicture} \hspace{1em}
                \begin{tikzpicture}
                        \begin{axis}[   x=3cm, y=1.5cm, axis lines=middle,  
                                xlabel=$t$,
                                ylabel=$h(t)$,
                                x label style={anchor=north},
                                y label style={anchor=east},
                                xmin=-1.2,xmax=1.2,ymin=-.2,ymax=1.4,
                                xtick={0,.3,.6}, ytick={1},
                                xticklabels={0, $T_s$, $2T_s$}, yticklabels = {$\frac{3}{2T_s}$},
                                clip=false,
                            ]
                        \addplot+[very thick,mark=none, teal] coordinates {(-1,0) (0,0) (0.3,1) (.6,0) (1,0)};
                        \end{axis}
                \end{tikzpicture}      
        \end{center}

        Por ser filtro acoplado, la relación señal a ruido a la salida es:
        \begin{equation*}
                \left(\frac{A}{\sigma}\right)^2_{\max} = 2 \frac{A^2}{\eta_0} ||p||^2 = \frac{4A^2T_s}{3\eta_0}.
        \end{equation*}

        Por ser unipolar binario, la probabilidad de error es entonces:
        \begin{equation*}
                P_e = Q\left(\frac{A}{2\sigma}\right) = Q\left(\sqrt{\frac{A^2 T_s}{3\eta_0}}\right)
        \end{equation*}

        Observación: el espectro del pulso es infinito en frecuencia, por lo que se requiere que el canal tenga suficiente ancho de banda como para no incorporar ISI (al menos $\pm f_s$ para dejar pasar el lóbulo central).

        \textbf{Parte 3.} El pulso transmitido vale $1$ en $t=0$ y $0$ en $t=\pm T_s,\pm 2T_s,\ldots$: cumple la condición de ISI nula. Pero a la salida del filtro acoplado el pulso es $r(t)=\kappa\,(p*p)(t-T_s)$, de duración $4T_s$, y en el instante de muestreo de un símbolo vecino vale
        \begin{equation*}
                \frac{(p*p)(T_s)}{(p*p)(0)}=\frac{\int_0^{T_s}\left(1-\frac{u}{T_s}\right)\frac{u}{T_s}\,du}{\frac23 T_s}=\frac{T_s/6}{2T_s/3}=\frac14,
        \end{equation*}
        mientras que $(p*p)(\pm 2T_s)=0$. La muestra es entonces $a_k+\frac{a_{k-1}+a_{k+1}}{4}$ más ruido: con símbolos unipolares, si $a_k=0$ y los dos vecinos valen $A$, la muestra sin ruido vale $A/2$, exactamente sobre el umbral. Ese patrón ocurre con probabilidad $1/8$ y da error con probabilidad $1/2$, por lo que $P_e\geqslant 1/16$ aunque no haya ruido. La fórmula de la parte 2 \emph{no} es válida: el filtro acoplado maximiza la relación señal a ruido, pero no garantiza ausencia de ISI. Con el pulso de duración $T_s$ del Ejercicio~\ref{ej:triangular corto}, en cambio, $p*p$ tiene duración $2T_s$, se anula en $\pm T_s$, y el filtro acoplado no introduce ISI.
\end{solucion}

\begin{ejercicio}
Un sistema de comunicación digital binario utiliza un canal con ruido blanco gaussiano de densidad espectral de potencia $\eta_0/2=10^{-15}\ \unit{\square\volt/\hertz}$. La velocidad de señalización es $1\times 10^6$ Baudios, y los pulsos son rectangulares NRZ.
\begin{enumerate}
\item Calcular la potencia necesaria para que la probabilidad de error de símbolo $P_e$ sea menor que $1\times 10^{-6}$ si se utiliza señalización unipolar. Recalcular para señalización polar.
\item ¿Cuál sería la potencia necesaria para mantener la misma velocidad de transmisión de información con igual probabilidad de error de \emph{bit} ($10^{-6}$, usando codificación de Gray) usando señalización M-aria, con una constelación de 8 símbolos de amplitudes $\{\pm A,\pm 3A,\pm 5A,\pm 7A\}$? Comparar en $\unit{\decibel}$ con el caso polar binario.
\end{enumerate}
\end{ejercicio}

\begin{solucion}
\begin{enumerate}
        \item Para el pulso NRZ, $||p||^2 = T_s = 1/f_s$ en este caso, y $\eta_0=2\times10^{-15}\ \unit{\square\volt/\hertz}$ (cuidado: el dato es $\eta_0/2$). De aquí se deduce que, usando filtro acoplado:
        \begin{equation*}
                \left(\frac{A}{\sigma}\right)^2_{\max} = 2 \frac{A^2}{\eta_0} ||p||^2 = \frac{2A^2}{f_s \eta_0},
        \end{equation*}
        y la probabilidad de error para el caso unipolar queda:
        \begin{equation*}
                P_e = Q\left(\frac{A}{2\sigma}\right)=Q\left(\sqrt{\frac{A^2}{2f_s \eta_0}}\right) = 10^{-6}.
        \end{equation*}
        De donde:
        \begin{equation*}
        \frac{A}{\sqrt{2f_s \eta_0}} \approx 4.753 \Rightarrow A \approx 4.753\sqrt{4\times10^{-9}}\ \unit{\volt}\approx 300\ \unit{\micro\volt}.
        \end{equation*}

        La energía media del pulso NRZ es $\bar{E}_p = \frac{A^2}{2}T_s = \frac{A^2}{2f_s}$ y la potencia total es $S = \bar{E}_p f_s = \frac{A^2}{2} \approx 4.5 \times 10^{-8}\ \unit{\watt}$ (sobre $1\ \unit{\ohm}$), es decir $\approx -43.4\ \mathrm{dBm}$.

        Si la codificación es polar binaria, la probabilidad de error es:
        \begin{equation*}
                P_e = Q\left(\frac{A}{\sigma}\right)=Q\left(\sqrt{\frac{2A^2}{f_s \eta_0}}\right) = 10^{-6}.
        \end{equation*}
        De donde:
        \begin{equation*}
        A\sqrt{\frac{2}{f_s \eta_0}} \approx 4.753 \Rightarrow A \approx 150\ \unit{\micro\volt}.
        \end{equation*}

        La energía media del pulso NRZ es $\bar{E}_p = A^2 T_s = \frac{A^2}{f_s}$ y la potencia total es $S = \bar{E}_p f_s = A^2 \approx 2.26 \times 10^{-8}\ \unit{\watt}\approx -46.5\ \mathrm{dBm}$ (la mitad, $3\ \unit{\decibel}$ menos).

        \item $\mathrm{vti} = 10^6 \unit{\bit/\sec}$, de donde podemos bajar $f_s$ a $f_s = \frac{\mathrm{vti}}{\log_2 M} \approx 333\ \unit{\kilo\baud}$. Con codificación de Gray, $\mathrm{BER}\approx P_e(\mathrm{sym})/\log_2 M$, por lo que la probabilidad de error de símbolo puede ser algo más alta, $P_e(\mathrm{sym}) = 3\times 10^{-6}$.

        Al ser un sistema $M-$ario polar, la probabilidad de error de símbolo es:
        \begin{equation*}
                P_e(\mathrm{sym}) = 2\frac{M-1}{M}Q\left(\frac{A}{\sigma}\right) = \frac{14}{8}Q\left(\frac{A}{\sigma}\right) = 3\times 10^{-6}
        \end{equation*}
        De donde se requiere un $A/\sigma \approx 4.643$. Al ser filtro acoplado, el $(A/\sigma)^2$ máximo es $2\frac{A^2}{\eta_0}||p||^2= 2\frac{A^2}{\eta_0 f_s}$ para el pulso NRZ. Por lo tanto:
        \begin{equation*}
        4.643^2 = 2\frac{A^2}{\eta_0 f_s} \Rightarrow A \approx 85\ \unit{\micro\volt}
        \end{equation*}

        La potencia transmitida es $\bar{E}_p f_s=\E{a_k^2}$, que en el caso $M-$ario polar queda:
        \begin{equation*}
                S = A^2 \frac{M^2-1}{3} =21A^2\approx 1.5\times 10^{-7}\ \unit{\watt} \approx -38.2\ \mathrm{dBm}.
        \end{equation*}
        Requiere unos $8.3\ \unit{\decibel}$ más de potencia que el polar binario, en un tercio del ancho de banda.
\end{enumerate}        
\end{solucion}

Un sistema de transmisión con señalización binaria polar, pulsos de Nyquist ideales, y 50 repetidores debe tener probabilidad de error menor que $10^{-4}$. \begin{ejercicio}
Sea la onda digital $x_T(t)= \sum_k a_k p_N(t-kT_S)$, con símbolos $a_k \in \{ 0, A, 2A\}$ equiprobables, y pulsos de Nyquist $p_N$ de coseno alzado, y exceso de ancho de banda $\alpha=0.8$.

\begin{enumerate}
\item Calcular la eficiencia espectral de la comunicación en $\unit{bps/\hertz}$.
\item Si se recibe esta señal contaminada de ruido blanco, diseñar en diagrama de bloques un detector apropiado indicando todos los detalles relevantes. 
\item Para el detector diseñado, dar una fórmula para la probabilidad de error $P_e$ de símbolo en función de $A$, $f_s$ y la densidad espectral del ruido en el canal $\eta_0$. Evaluar $P_e$ para el caso de densidad espectral de ruido $\eta_0/2=10^{-15} \unit{\square\volt/\hertz}$, amplitud $A=0,6\,\unit{\milli\volt}$, y una velocidad de transmisión de información de 4 \unit{\mega\bit/\sec}.
\end{enumerate}
\end{ejercicio}

\begin{solucion}
\begin{enumerate}
        \item El pulso $p_N$ requiere un ancho de banda $B = \frac{f_s}{2}(1+\alpha) = 0.9f_s$. La $\mathrm{vti} = \log_2(3)f_s$ de donde la eficiencia queda:
        \begin{equation*}
                \mathrm{Eff} = \frac{\mathrm{vti}}{B} = \frac{\log_2(3)}{0.9} \approx 1.76.
        \end{equation*}

        \item Como el pulso transmitido ya es de Nyquist, el filtro acoplado \emph{no} es una buena opción: el pulso a su salida tiene espectro $|\hat P_N(f)|^2$, el cuadrado de un coseno alzado, que no tiene simetría vestigial, y se introduce ISI. Para $\alpha=0.8$, la muestra de un pulso vecino a la salida del filtro acoplado vale aproximadamente $16\%$ de la muestra útil, lo que con umbrales separados $A$ no es despreciable.

        El detector apropiado (ver el ejemplo de pulsos de Nyquist no ideales en las notas) es un filtro pasabajos ideal de ancho $B=\frac{f_s}{2}(1+\alpha)=0.9f_s$, que deja pasar el pulso intacto, seguido de un muestreador en $kT_s$ y un comparador con umbrales $A/2$ y $3A/2$. (La alternativa óptima, mencionada en las notas, es repartir el coseno alzado entre transmisor y receptor con pulsos de raíz de coseno alzado; pero aquí el pulso transmitido está dado.)

        \item Para el sistema de detección anterior, la probabilidad de error de símbolo es:
        \begin{equation*}
                P_e(\mathrm{sym}) = \frac{1}{3}Q\left(\frac{A}{2\sigma}\right) + \frac{1}{3}2Q\left(\frac{A}{2\sigma}\right) + \frac{1}{3} Q\left(\frac{A}{2\sigma}\right) = \frac{4}{3}Q\left(\frac{A}{2\sigma}\right)
        \end{equation*}

        Con el pasabajos ideal de ancho $B$, la varianza del ruido es $\sigma^2=\frac{\eta_0}{2}\cdot 2B=\frac{\eta_0 f_s}{2}(1+\alpha)$, de donde
        \begin{equation*}
                P_e(\mathrm{sym}) =  \frac{4}{3} Q\left(\sqrt{\frac{A^2}{2\eta_0(1+\alpha)f_s}}\right).
        \end{equation*}

        Para los valores dados, se requiere $\textrm{vti} = 4\times 10^6$ por lo que $f_s = \textrm{vti}/\log_2(3) \approx 2.52 \unit{\mega\hertz}$, y con $\eta_0=2\times10^{-15}\ \unit{\square\volt/\hertz}$:
        \begin{equation*}
                \sqrt{\frac{A^2}{2\eta_0(1+\alpha) f_s}} \approx 4.45 \Rightarrow P_e \approx \frac43 Q(4.45) \approx 5.7 \times 10^{-6}.
        \end{equation*}

        Como comparación, si se ignorara la ISI del filtro acoplado, con $\|p_N\|^2=T_s(1-\alpha/4)$ se tendría $\left(\frac{A}{2\sigma}\right)^2=\frac{A^2\|p_N\|^2}{2\eta_0}=\frac{0.4A^2}{\eta_0f_s}$ y $P_e\approx\frac43Q(5.34)\approx 6\times10^{-8}$. Esa es la cota que alcanzaría un sistema con pulsos de raíz de coseno alzado y la misma energía de pulso.

\end{enumerate}
\end{solucion}

\end{equation*}
y encuentre una expresión para la probabilidad de error $P_e$.

\item Supongamos que los pulsos $p(t)$ son pulsos de Nyquist de coseno alzado con exceso de ancho de banda 100\%. La señalización es polar con $A=500 \unit{\micro\volt}$ y velocidad $10 \unit{\mega\bit/\sec}$. Además, el canal presenta ruido blanco Gaussiano aditivo con densidad espectral de potencia $\eta_0/2~=~5\times 10^{-16} \,\unit{\volt}^2/\unit{Hz}$.

\begin{enumerate}
\item Si para la detección digital se utiliza un filtro acoplado, calcule la varianza del ruido a la salida del filtro. ¿Cuál es la desventaja de utilizar este tipo de filtro?

\item Proponga un filtro de recepción alternativo que solucione el inconveniente anterior. Utilizando el umbral óptimo de la parte 1, calcule la probabilidad de error $P_e$, para el caso $P(A_1)=\frac{1}{3}$, $P(A_2)=\frac{2}{3}$,

\end{enumerate}
\end{enumerate}

\begin{solucion}
\begin{enumerate}
\item La señal contaminada con ruido tiene la siguiente densidad, mezcla de dos Gaussianas:
        \begin{equation*}
                f_X(x) = P(A_1) \frac{1}{\sqrt{2\pi}\sigma}e^{-\frac{(x-A_1)^2}{2\sigma^2}} + P(A_2)\frac{1}{\sqrt{2\pi}\sigma}e^{-\frac{(x-A_2)^2}{2\sigma^2}}.
        \end{equation*}
        \begin{center}
        \begin{tikzpicture}
                \pgfmathdeclarefunction{gauss}{3}{%
                \pgfmathparse{1/(#3*sqrt(2*pi))*exp(-((#1-#2)^2)/(2*#3^2))}%
                }

                        ylabel=$f_X(x)$,

                        xticklabels={$A_1$, $A^*$, $A_2$},

                    ]
                \addplot+[dashed ,mark=none, red, domain=(-1:4), samples=100] {.3*gauss(x,1,.5)};
                \addplot+[dashed ,mark=none, red, domain=(-1:4), samples=100] {.7*gauss(x,3,.5)};
                \addplot+[very thick,mark=none, teal, domain=(-1:4), samples=100] {.3*gauss(x,1,.5)+.7*gauss(x,3,.5)};
                \end{axis}
        \end{tikzpicture}
        \end{center}
        El umbral óptimo es cuando se igualan las densidades de cada símbolo, es decir:
        \begin{equation*}
                \frac{P(A_2)}{P(A_1)} = \frac{e^{-\frac{(A^*-A_1)^2}{2\sigma^2}}}{e^{-\frac{(A^*-A_2)^2}{2\sigma^2}}} = e^{\frac{(A^*-A_2)^2-(A^*-A_1)^2}{2\sigma^2}}.
        \end{equation*}
        Por lo tanto:
        \begin{equation*}
                2\sigma^2 \log\left(\frac{P(A_2)}{P(A_1)}\right) = (A^*-A_2)^2-(A^*-A_1)^2 = (2A^*-A_1-A_2)(A_1-A_2),
        \end{equation*}
        de donde despejando $A^*$ se obtiene la fórmula pedida.

        La probabilidad de error es:
        \begin{equation*}
                P_e = P(A_1)Q\left(\frac{A^*-A_1}{\sigma}\right)+ P(A_2)Q\left(\frac{A_2-A^*}{\sigma}\right)
        \end{equation*}

\item   \begin{enumerate}
        \item En este caso es señalización polar $\pm A$ equiprobable, por lo que el umbral es $0$ y por lo tanto, al usar filtro acoplado tenemos que en recepción:
        \begin{equation*}
                \left(\frac{A}{\sigma}\right)^2_{\max} = 2 \frac{A^2}{\eta_0} ||p||^2
        \end{equation*}

        \begin{equation*}
                \sigma^2=\frac{\eta_0}{2}\|h\|^2=\frac{\eta_0}{2\|p\|^2}=\frac{5\times10^{-16}}{0.75\times10^{-7}}\approx 6.7\times10^{-9}\ \unit{\square\volt},
        \end{equation*}
        y, si no hubiera ISI, la probabilidad de error sería:
        \begin{equation*}
                Q\left(\frac{A}{\sigma}\right) = Q\left(\sqrt{\frac{3A^2}{2\eta_0 f_s}}\right) \approx Q(6.12) \approx 4.6 \times 10^{-10}.
        \end{equation*}

        \item El problema del filtro acoplado es que introduce ISI (para $\alpha=1$, cada vecino aporta $1/6$ de su amplitud en el instante de muestreo), que en este caso puede ser peor que el ruido. Por lo tanto, podemos usar un filtro pasabajos ideal de ancho $f_s$, el ancho de banda total del pulso. En ese caso, la onda pasará intacta sin ISI y al muestrearla en los lugares correctos obtenemos la densidad de la parte $1$ si las probabilidades son distintas. En este caso, la varianza del ruido es $\sigma^2 = \eta_0 f_s = 10^{-8}$ ya que no es acoplado (hay exceso de ruido).
        
        Calculando para $A_1=-A$, $A_2=A$ y $P(A_1)=\frac{1}{3}=1-P(A_2)$ obtenemos:
        \begin{equation*}
                A^* \approx - 6.93\times 10^{-6}, \quad P_e = 2.7\times 10^{-7}
        \end{equation*}

        El que sean probabilidades diferentes no afecta mucho la probabilidad de error, el problema es el filtro no acoplado. Si fueran equiprobables los símbolos la prob. de error sería $2.86\times 10^{-7}$, bastante mayor que en la parte anterior.
        \end{enumerate}
\end{enumerate}
\end{solucion}

\begin{ejercicio}\label{ej:triangular corto}

Se considera un canal digital de banda base contaminado con ruido blanco de densidad espectral $\eta_0/2 = 10^{-5}\,\unit{\square\volt/\hertz}$. Se utiliza señalización polar con símbolos equiprobables de amplitud $A=3.0\,V$ en la recepción, y velocidad de señalización $f_s=10\,\unit{\kilo\baud}$. Se utilizan pulsos triangulares de la forma:
\begin{equation*}
        p(t)=\begin{cases}
                1-\frac{2|t|}{T_s} &\text{ para } |t|\leq \frac{T_s}{2}\\
                0 & \text{ en otro caso}
            \end{cases}
\end{equation*}
\begin{enumerate}
\item Determinar la mínima probabilidad de error en la detección, posible del sistema. ¿Cómo se obtiene dicho valor?
\item La comunicación anterior se repite en 10 tramos
idénticos, con la amplificación necesaria para restituir el nivel
de señal. Determinar la probabilidad de error a la salida si los
amplificadores: 
\begin{enumerate}
\item no son regenerativos
\item son regenerativos
\end{enumerate}
\end{enumerate}
\end{ejercicio}

\begin{solucion}
        \begin{enumerate}
                \item Si se usa filtro acoplado y señalización polar, la relación a la salida es:
                \begin{equation*}
                        \left(\frac{A}{\sigma}\right)^2_{\max} = 2 \frac{A^2}{\eta_0}||p||^2
                \end{equation*}
                Para este pulso $||p||^2$ se puede obtener integrando en el tiempo y da $||p||^2 = T_s/3$. Como la señalización es polar, el umbral es $0$ y por lo tanto:
                \begin{equation*}
                        P_e = Q\left(\sqrt{\frac{2A^2}{3\eta_0 f_s}}\right) \approx Q(5.48) \approx 2.2\times 10^{-8}.
                \end{equation*}

                \item \begin{enumerate}
                        \item Si no son regenerativos, la $(A/\sigma)^2_{\max}$ se degrada por $m$ por lo que:
                        \begin{equation*}
                                P_e = Q\left(\sqrt{\frac{2A^2}{(10) 3\eta_0 f_s}}\right) \approx Q(1.73) \approx 0.041.
                        \end{equation*}
                        \item Si son regenerativos, la probabilidad de error es $\approx m P_e \approx 2.2 \times 10^{-7}$.
                \end{enumerate}
        \end{enumerate}
\end{solucion}


\section{Modulación digital pasabanda}

\begin{ejercicio}\label{ej:bpsk}
Un sistema de señalización binaria BPSK trasmite información a una velocidad de $2.5 \unit{\mega\bit/\second}$. El ruido en el canal es blanco y gaussiano de densidad espectral de potencia $\eta_0/2=10^{-20}\unit{\square\volt/\hertz}$. La amplitud de la señal a la entrada del detector es $A=1\unit{\micro\volt}$. Asumiendo que se trata de detección coherente con filtro acoplado, determinar la probabilidad de error de bit.
\end{ejercicio}

\begin{solucion}
        La detección coherente BPSK se transforma en banda base a detectar $\pm A$ con umbral $0$. El ruido demodulado tiene densidad $\eta_0$ (el doble que en banda base), y con el filtro acoplado normalizado (promedio en un tiempo de símbolo) su varianza es $\sigma^2 = \eta_0\|h\|^2=\eta_0 f_s$, de donde:
        \begin{equation*}
                P_e = Q\left(\frac{A}{\sigma}\right)=Q\left(\sqrt{\frac{A^2}{\eta_0 f_s}}\right) =Q\left(\sqrt{\frac{10^{-12}}{2\times10^{-20}\cdot 2.5\times10^{6}}}\right)= Q(\sqrt{20})\approx Q(4.47) \approx 3.9\times 10^{-6}.
        \end{equation*}
\end{solucion}

\begin{ejercicio}[Ruido en fase y cuadratura]
Con los datos del Ejercicio~\ref{ej:bpsk} (pulsos NRZ, $f_s=2.5\ \unit{\mega\baud}$), el receptor tiene primero un filtro pasabanda ideal de ancho $2f_s$ centrado en $f_c$, y luego los mezcladores con $2\cos(2\pi f_c t)$ y $-2\sin(2\pi f_c t)$, seguidos de filtros pasabajos ideales de ancho $f_s$. Suponer que los filtros dejan pasar la señal sin modificar apreciablemente su potencia.
\begin{enumerate}
\item Calcular la potencia del ruido pasabanda $n(t)$ a la salida del primer filtro y la relación señal a ruido a la entrada de los mezcladores.
\item Usando el resultado de las notas sobre el ruido demodulado, dar la densidad espectral de $n_i(t)$ y $n_q(t)$ y su potencia a la salida de los pasabajos. Verificar que la potencia de $n(t)$ es el promedio de las potencias de $n_i(t)$ y $n_q(t)$.
\item Calcular la relación señal a ruido en la rama en fase y compararla con la de la parte 1. ¿De dónde sale la diferencia? ¿Qué hay en la rama en cuadratura?
\item Repetir las partes 1 y 3 para una señal $4$-QAM con la misma amplitud $A$ en cada rama, $a_k,b_k\in\{-A,A\}$. ¿Se mantiene la ganancia de la parte 3? ¿Qué se gana a cambio?
\item Comparar la relación señal a ruido de la parte 3 con la relación $(A/\sigma)^2$ a la salida del filtro acoplado, y recuperar el resultado del Ejercicio~\ref{ej:bpsk}. ¿De dónde sale la diferencia?
\end{enumerate}
\end{ejercicio}

\begin{solucion}
Tenemos $A=1\ \unit{\micro\volt}$, $\eta_0=2\times10^{-20}\ \unit{\square\volt/\hertz}$ y $f_s=2.5\times10^6\ \unit{\baud}$.
\begin{enumerate}
\item El filtro deja pasar el ruido en dos bandas de ancho $2f_s$, alrededor de $\pm f_c$, con densidad $\eta_0/2$:
\begin{equation*}
N=\frac{\eta_0}{2}\cdot 2\cdot 2f_s=2\eta_0f_s=10^{-13}\ \unit{\square\volt}.
\end{equation*}
La señal BPSK, $x_c(t)=\sum_ka_kp(t-kT_s)\cos(2\pi f_ct)$, tiene potencia $\frac{A^2}{2}=5\times10^{-13}\ \unit{\square\volt}$, de donde $\mathrm{SNR}_{\text{pb}}=5$, es decir $7\ \unit{\decibel}$.
\item Escribiendo $n(t)=n_i(t)\cos(2\pi f_ct)-n_q(t)\sin(2\pi f_ct)$, la proposición de las notas dice que $n_i$ y $n_q$ son independientes y tienen densidad espectral $\eta_0$ en $|f|\leqslant f_s$ (la mitad del ancho del pasabanda). Sus potencias son entonces $P_{n_i}=P_{n_q}=\eta_0\cdot2f_s=10^{-13}\ \unit{\square\volt}$. Como $\cos^2$ y $\sin^2$ tienen valor medio $\frac12$ y $n_i$, $n_q$ son independientes,
\begin{equation*}
P_n=\frac{P_{n_i}+P_{n_q}}{2}=10^{-13}\ \unit{\square\volt},
\end{equation*}
que coincide con la parte 1.
\item En la rama en fase se tiene $\sum_ka_kp(t-kT_s)+n_i(t)$. La señal tiene potencia $A^2=10^{-12}\ \unit{\square\volt}$ (el factor $2$ del oscilador recupera la amplitud $A$) y el ruido $10^{-13}\ \unit{\square\volt}$, por lo que $\mathrm{SNR}_i=10$, es decir $10\ \unit{\decibel}$: $3\ \unit{\decibel}$ mejor que la pasabanda.

El factor $2$ del oscilador no explica la diferencia, porque multiplica por igual a la señal y al ruido. La diferencia se debe a que la potencia del ruido pasabanda se reparte por igual entre $n_i$ y $n_q$, mientras que toda la potencia de la señal BPSK está en fase. En la rama en cuadratura hay \emph{solo ruido}, $n_q(t)$, y el detector coherente lo descarta. Ese es el origen de los $3\ \unit{\decibel}$.
\item Para $4$-QAM la potencia de la señal pasabanda es $\frac{\E{a_k^2}+\E{b_k^2}}{2}=A^2=10^{-12}\ \unit{\square\volt}$, por lo que $\mathrm{SNR}_{\text{pb}}=10$ ($10\ \unit{\decibel}$). En cada rama se tiene una señal de potencia $A^2$ con ruido de potencia $10^{-13}\ \unit{\square\volt}$: $\mathrm{SNR}_i=\mathrm{SNR}_q=10$. Ahora no hay ganancia respecto de la SNR pasabanda, porque la rama en cuadratura también lleva señal y no se descarta nada.

A cambio, se transmiten $2$ bits por símbolo en el mismo ancho de banda. Como cada rama es un BPSK con la misma amplitud y el mismo ruido, la probabilidad de error de bit es la misma que en BPSK, con el doble de tasa (y el doble de potencia, es decir la misma energía por bit). BPSK desperdicia la dimensión en cuadratura; $4$-QAM la usa.
\item A la salida del filtro acoplado normalizado, $\sigma^2=\eta_0\|h\|^2=\eta_0f_s=5\times10^{-14}\ \unit{\square\volt}$, y $(A/\sigma)^2=20$ ($13\ \unit{\decibel}$), de donde $P_e=Q(\sqrt{20})\approx3.9\times10^{-6}$, como en el Ejercicio~\ref{ej:bpsk}.

La diferencia de $3\ \unit{\decibel}$ con la parte 3 se debe al ancho de banda equivalente de ruido. El pasabajos ideal de ancho $f_s$ deja pasar el ruido en $2f_s$ (frecuencias positivas y negativas), mientras que para el integrador $\int|\hat H(f)|^2df=\int\sinc^2(\pi fT_s)\,df=f_s$, la mitad. El filtro acoplado deja pasar menos ruido conservando el valor de la muestra.
\end{enumerate}
\end{solucion}

\begin{ejercicio}[Envolvente compleja, error de fase y error de frecuencia]
Se transmite una señal $4$-QAM,
\begin{equation*}
x_c(t)=\sum_k \left[a_k\cos(2\pi f_c t)-b_k\sin(2\pi f_c t)\right]p(t-kT_s),\qquad a_k,b_k\in\{-A,A\},
\end{equation*}
con pulsos de raíz de coseno alzado con $\alpha=0.25$, velocidad de señalización $f_s=4000\ \unit{\baud}$ y portadora $f_c=16\ \unit{\kilo\hertz}$. El receptor es el detector coherente con filtros acoplados, y el canal agrega ruido blanco gaussiano de densidad espectral $\eta_0/2$.
\begin{enumerate}
\item Escribir la envolvente compleja $x_{ce}(t)$ y verificar que $x_c(t)=\mathrm{Re}\left[x_{ce}(t)\e^{j2\pi f_c t}\right]$. ¿Qué banda de frecuencias ocupa $x_c(t)$?
\item \emph{Error de fase.} Los osciladores locales del receptor son $2\cos(2\pi f_c t+\theta)$ y $-2\sin(2\pi f_c t+\theta)$. Mostrar que, a la salida de los pasabajos, la envolvente compleja demodulada es $x_{ce}(t)\e^{-j\theta}$, es decir, la constelación aparece rotada un ángulo $-\theta$. ¿Cambia la potencia del ruido demodulado?
\item Para el caso BPSK ($b_k=0$), hallar la probabilidad de error en función de $\theta$ y de $\gamma=\bar E_b/\eta_0$. ¿Cuántos $\unit{\decibel}$ de energía de bit adicional se necesitan para compensar un error de fase de $20^\circ$? ¿Qué pasa si $\theta=\pi$?
\item Para el caso $4$-QAM, mostrar que la muestra de la rama en fase es $a_k\cos\theta+b_k\sin\theta$ (más ruido): las dos ramas se mezclan. Hallar la probabilidad de error de la rama en fase en función de $\theta$ y $\gamma$, y la penalización en $\unit{\decibel}$ del peor caso para $\theta=20^\circ$. Comparar con BPSK. ¿A partir de qué $\theta$ el detector se equivoca aun sin ruido?
\item \emph{Error de frecuencia.} Si los osciladores locales están en $f_c+\Delta f$, mostrar que la envolvente compleja demodulada es $x_{ce}(t)\e^{-j2\pi\Delta f\,t}$. Para $\Delta f=10\ \unit{\hertz}$, ¿cuántos grados gira la constelación en cada símbolo? ¿Cada cuánto tiempo da una vuelta completa? Despreciando el ruido, ¿a partir de qué símbolo se equivoca el detector en BPSK y en $4$-QAM? ¿Qué mecanismo resuelve este problema?
\item En el caso anterior, $\Delta f$ es $625$ ppm de la portadora. Una radio real a $2.4\ \unit{\giga\hertz}$ usa en cada extremo osciladores de cristal con una precisión de $\pm 20$ ppm. Calcular el máximo $\Delta f$ y cuántas vueltas giraría la constelación en cada símbolo a $4000\ \unit{\baud}$. Comparar $\Delta f$ con el ancho de banda de la señal: ¿qué se obtiene a la salida del filtro acoplado? ¿Alcanza con el mecanismo de la parte anterior?
\end{enumerate}
\end{ejercicio}

\begin{solucion}
\begin{enumerate}
\item $x_{ce}(t)=\sum_k(a_k+jb_k)\,p(t-kT_s)$. En efecto,
\begin{equation*}
\mathrm{Re}\left[(a_k+jb_k)\left(\cos(2\pi f_ct)+j\sin(2\pi f_ct)\right)\right]=a_k\cos(2\pi f_ct)-b_k\sin(2\pi f_ct).
\end{equation*}
El espectro del pulso de raíz de coseno alzado está contenido en $|f|\leqslant\frac{(1+\alpha)f_s}{2}=2.5\ \unit{\kilo\hertz}$, por lo que $x_c(t)$ ocupa la banda de $13.5$ a $18.5\ \unit{\kilo\hertz}$: $5\ \unit{\kilo\hertz}$ centrados en $f_c$.

\item Las dos ramas del receptor forman la señal compleja
\begin{equation*}
z(t)=x_c(t)\left[2\cos(2\pi f_ct+\theta)-j2\sin(2\pi f_ct+\theta)\right]=2x_c(t)\e^{-j(2\pi f_ct+\theta)}.
\end{equation*}
Como $x_c=\frac12\left(x_{ce}\e^{j2\pi f_ct}+\overline{x_{ce}}\e^{-j2\pi f_ct}\right)$,
\begin{equation*}
z(t)=x_{ce}(t)\e^{-j\theta}+\overline{x_{ce}(t)}\e^{-j(4\pi f_ct+\theta)},
\end{equation*}
y los pasabajos eliminan el segundo término, centrado en $-2f_c$. Queda $x_{ce}(t)\e^{-j\theta}$: cada símbolo $a_k+jb_k$ aparece multiplicado por $\e^{-j\theta}$, es decir rotado un ángulo $-\theta$.

Con el ruido ocurre lo mismo: se obtiene $(n_i+jn_q)\e^{-j\theta}$, cuya parte real es $n_i\cos\theta+n_q\sin\theta$. Como $n_i$ y $n_q$ son independientes y de igual densidad $\eta_0$, esta combinación tiene densidad $\eta_0(\cos^2\theta+\sin^2\theta)=\eta_0$. El error de fase no cambia el ruido; solo rota la señal.

\item Para BPSK basta con la rama en fase:

\begin{blockdiagram}[decoration={brace,amplitude=10}]
        \matrix[row sep=5 mm, column sep=8mm, ampersand replacement=\&]{
            \& \& \& \& \& \node[punto] (p1) {}; \& \node[etiqueta] (muestreo) {};\\
            \node [etiqueta] (p6) {$x_c(t)+n(t)$}; \& \node[operator] (mult) {$\times$}; \& \node [block] (filtro) {$\displaystyle \frac{1}{T_s} \int_{t-T_s}^t$}; \& \node [punto] (p2) {}; \&\node [punto] (p3) {}; \& \&\node [punto] (p4) {}; \&\node [block] (comp) {COMP};\\
            \& \node [etiqueta] (osc) {$2\cos(\omega_ct + \theta)$}; \& \& \& \node [block] (sinc) {SINCR}; \& \node [punto] (p5) {};\\
        };
        \draw [-stealth] (p6.east) --  (mult);
        \draw [-stealth] (mult) -- node [anchor=south] {$x_o(t)$} (filtro);
        \draw [-stealth] (osc) -- (mult);
        %\draw (filtro) -- (p2) -- (p3) -- (muestreo);
        \draw (filtro) -- (p2) -- node [anchor=south] {$y$} (p3) -- (muestreo);
        \draw [-stealth] (p2) |- (sinc);
        \draw (sinc) -- (p5);
        \draw [-stealth,dashed] (p5) -- (p1);
        \draw [-stealth] (p4) -- (comp);
        %\draw [decorate]  (filtro.south east) --node [below=3mm] {\parbox{3cm}{filtro acop. a NRZ, integra en un período de símb.}} (filtro.south west);
    \end{blockdiagram}

con $x_c(t)=\sum_ka_kp(t-kT_s)\cos(2\pi f_ct)$. Por la parte anterior, la salida de la rama en fase es $\sum_ka_kp(t-kT_s)\cos\theta+n_i(t)\cos\theta+n_q(t)\sin\theta$, con ruido de densidad $\eta_0$. La amplitud del símbolo queda multiplicada por $\cos\theta$, y con el filtro acoplado
\begin{equation*}
\left(\frac{A\cos\theta}{\sigma}\right)^2_{\max}=\frac{A^2\cos^2\theta}{\eta_0}\|p\|^2=2\gamma\cos^2\theta,
\end{equation*}
usando $\bar E_b=A^2\|p\|^2/2$. Entonces
\begin{equation*}
P_e=Q\left(\sqrt{2\gamma}\,\cos\theta\right).
\end{equation*}
El error de fase equivale a perder un factor $\cos^2\theta$ de energía: para $\theta=20^\circ$ hay que aumentar $\bar E_b$ en $-20\log_{10}\cos20^\circ\approx0.54\ \unit{\decibel}$.

Si $\theta=\pm\pi/2$, la rama en fase solo tiene ruido y $P_e=1/2$. Si $|\theta|>\pi/2$, $\cos\theta$ cambia de signo y los símbolos se invierten. En el caso extremo $\theta=\pi$, $P_e=1-Q(\sqrt{2\gamma})\approx1$: se decodifica todo el mensaje al revés. Es la ambigüedad de fase que dejan abierta el lazo cuadrático y el lazo de Costas (capítulo de sincronización).

\item $(a_k+jb_k)\e^{-j\theta}=(a_k\cos\theta+b_k\sin\theta)+j(b_k\cos\theta-a_k\sin\theta)$. La muestra de la rama en fase es $a_k\cos\theta+b_k\sin\theta$ más ruido: además de atenuarse, recibe la interferencia de la otra rama.

Para $0\leqslant\theta<\pi/4$ y $a_k=A$, la muestra sin ruido vale $A(\cos\theta+\sin\theta)$ si $b_k=A$ y $A(\cos\theta-\sin\theta)$ si $b_k=-A$, con igual probabilidad (el caso $a_k=-A$ es simétrico). Para $4$-QAM cada rama tiene $(A/\sigma)^2=2\gamma$, igual que BPSK, por lo que
\begin{equation*}
P_e(a_k)=\frac12Q\left(\sqrt{2\gamma}\,(\cos\theta-\sin\theta)\right)+\frac12Q\left(\sqrt{2\gamma}\,(\cos\theta+\sin\theta)\right).
\end{equation*}
Para $\gamma$ grande domina el primer término, y el peor caso pierde un factor $(\cos\theta-\sin\theta)^2$ de energía. Para $\theta=20^\circ$, $\cos20^\circ-\sin20^\circ\approx0.598$, es decir $-20\log_{10}0.598\approx4.5\ \unit{\decibel}$, contra $0.54\ \unit{\decibel}$ en BPSK. $4$-QAM es mucho más sensible al error de fase: los puntos de la constelación están a $45^\circ$ de los bordes de las regiones de decisión, y no a $90^\circ$ como en BPSK.

Para $\theta\geqslant45^\circ$, $\cos\theta-\sin\theta\leqslant0$ y el detector se equivoca aun sin ruido.

\item Repitiendo la parte 2 con $2\cos(2\pi(f_c+\Delta f)t)$, es decir con $\theta=2\pi\Delta f\,t$, la envolvente demodulada es $x_{ce}(t)\e^{-j2\pi\Delta f\,t}$: la constelación gira continuamente con velocidad angular $2\pi\Delta f$.

En cada símbolo gira $360^\circ\cdot\Delta f\,T_s=360^\circ\cdot10/4000=0.9^\circ$, y da una vuelta completa cada $1/\Delta f=0.1\ \unit{\second}$, es decir cada $400$ símbolos. (El giro dentro de un mismo símbolo es despreciable, por lo que el filtro acoplado sigue funcionando bien en cada símbolo.)

Sin ruido, BPSK se equivoca cuando el giro supera $90^\circ$, a partir del símbolo $100$ ($25\ \unit{\milli\second}$). A partir de allí decodifica invertido durante $200$ símbolos, vuelve a acertar, y así sucesivamente: la amplitud en fase oscila como $A\cos(2\pi\Delta f\,t)$. $4$-QAM se equivoca cuando el giro supera $45^\circ$, a partir del símbolo $50$ ($12.5\ \unit{\milli\second}$). Antes, ya en el símbolo $22$ el giro es de $20^\circ$ y, por la parte anterior, se pierden $4.5\ \unit{\decibel}$ de margen.

El problema se resuelve con la sincronización de portadora: un PLL (con un detector de fase apropiado, como el lazo de Costas) que ajusta continuamente la fase del oscilador local. Para seguir un error de frecuencia sin error de fase en régimen se necesita un lazo de segundo orden.

\item $\Delta f_{\max}=2\times20\times10^{-6}\times2.4\times10^9=96\ \unit{\kilo\hertz}$ (los dos osciladores se apartan en sentidos opuestos). A $4000\ \unit{\baud}$, eso es $96000/4000=24$ vueltas por símbolo.

La envolvente demodulada queda desplazada $96\ \unit{\kilo\hertz}$ en frecuencia, mientras que el filtro acoplado deja pasar solo $|f|\leqslant2.5\ \unit{\kilo\hertz}$. La señal cae completamente fuera de la banda del filtro, y a su salida solo hay ruido. Ya no se trata de una constelación que gira lentamente: no hay constelación.

El PLL no alcanza, porque necesita que la señal esté presente a la salida de los filtros para medir la fase, y porque su rango de enganche es mucho menor que $96\ \unit{\kilo\hertz}$. Antes hay que estimar y corregir groseramente la frecuencia (por ejemplo, a partir de un preámbulo conocido o buscando la portadora en el espectro), y dejar al PLL solo el error residual. En el ejemplo de la parte anterior, $625$ ppm a $16\ \unit{\kilo\hertz}$ son solo $10\ \unit{\hertz}$; el problema real aparece porque el error en ppm se multiplica por una portadora de $\unit{\giga\hertz}$ mientras que la velocidad de señalización sigue siendo baja.
\end{enumerate}
\end{solucion}

\begin{ejercicio}[OOK frente a BPSK]
Se comparan dos sistemas binarios con pulsos NRZ, la misma velocidad de señalización y ruido blanco gaussiano de densidad espectral $\eta_0/2$: OOK, con $a_k\in\{0,A\}$, y BPSK, con $a_k\in\{-A',A'\}$, ambos con detección coherente y filtro acoplado.
\begin{enumerate}
\item A partir del espectro de OOK visto en las notas, calcular la potencia media de la señal de RF y la fracción de esa potencia que está en la componente discreta en $f_c$ (la portadora). Repetir para BPSK.
\item Hallar para cada sistema el valor de $\gamma=\bar E_b/\eta_0$, en $\unit{\decibel}$, necesario para una probabilidad de error de bit de $10^{-6}$. Interpretar la diferencia a partir del resultado de la parte anterior.
\item ¿Cuánto debe valer $A'$ en función de $A$ para que los dos sistemas tengan la misma potencia media? ¿Y para que tengan la misma probabilidad de error?
\item La portadora de OOK, ¿puede aprovecharse directamente para sincronizar el oscilador local del receptor? ¿Y en BPSK? Mencionar una ventaja práctica de OOK que justifique su uso en aplicaciones sencillas, como los controles remotos.
\end{enumerate}
\end{ejercicio}

\begin{solucion}
\begin{enumerate}
\item De las notas, la onda equivalente en banda base de OOK tiene $S_x(f)=\frac{A^2T_s}{4}\sinc^2(\pi T_sf)+\frac{A^2}{4}\delta(f)$, y la de RF es $S_{x_c}(f)=\frac14\left[S_x(f-f_c)+S_x(f+f_c)\right]$. En cada una de las dos bandas hay una parte continua de potencia $\frac{A^2}{16}$ y una delta de peso $\frac{A^2}{16}$. La potencia total es $4\cdot\frac{A^2}{16}=\frac{A^2}{4}$ (que coincide con $\E{a_k^2}/2$), y las dos deltas suman $\frac{A^2}{8}$: la portadora se lleva el $50\%$ de la potencia.

Para BPSK, $\mu_a=0$: no hay componente discreta, y toda la potencia, $\frac{A'^2}{2}$, está en la parte continua, que es la que lleva la información.
\item De las notas, $P_e^{\text{OOK}}=Q(\sqrt{\gamma})$ y $P_e^{\text{BPSK}}=Q(\sqrt{2\gamma})$. Con $Q^{-1}(10^{-6})\approx4.75$ se obtiene $\gamma\approx22.6$ ($13.5\ \unit{\decibel}$) para OOK y $\gamma\approx11.3$ ($10.5\ \unit{\decibel}$) para BPSK.

Los $3\ \unit{\decibel}$ de diferencia son exactamente la mitad de la potencia que OOK pone en la portadora, que no transporta información. Dicho de otro modo: si a la señal OOK se le restara su media $A/2$, quedaría una señal BPSK de amplitud $A/2$, con la misma distancia $A$ entre símbolos y la mitad de la potencia.
\item Igual potencia: $\frac{A'^2}{2}=\frac{A^2}{4}$, es decir $A'=A/\sqrt2$. Igual probabilidad de error: igual distancia entre símbolos, $2A'=A$, es decir $A'=A/2$, con la mitad de la potencia.
\item Sí: la componente discreta en $f_c$ puede extraerse con un filtro angosto o seguirse directamente con un PLL. BPSK es de portadora suprimida, y necesita un lazo cuadrático o de Costas (capítulo de sincronización). La ventaja práctica de OOK es que puede detectarse de forma \emph{no coherente}, con un detector de envolvente, sin oscilador local ni sincronismo de portadora, y que el transmisor solo debe encender y apagar un oscilador. Ambas cosas abaratan mucho el equipo, a cambio de los $3\ \unit{\decibel}$ (y algo más con detección no coherente).
\end{enumerate}
\end{solucion}

\begin{ejercicio}
Considere el sistema QPSK cuya señal transmitida es
\begin{equation*}
x_c(t)= \sum_k A p_R(t-kT_s)\cos(2\pi f_c t + \theta_k ),\quad\quad \theta_k\in\left\{0,\frac{\pi}{2},\pi,\frac{3\pi}{2}\right\}.
\end{equation*}
Aquí $p_R(t)$ es el pulso rectangular NRZ.
\begin{enumerate}
\item Se dispone de dos generadores locales, $\cos(\omega_c t+\pi/4)$ y $\cos(\omega_c t+3\pi/4)$. Diseñar en diagrama de bloques un detector sincrónico
apropiado usando filtros acoplados.
\item Si el canal agrega ruido blanco gaussiano de densidad espectral $\eta_0/2$, calcular la probabilidad de error de símbolo del detector diseñado y expresarla en función de la relación
$\gamma_s:=\frac{\bar{E}_p}{\eta_0}$, siendo $\bar E_p$ la energía media del pulso de RF transmitido. Expresarla también en función de $\gamma=\bar E_b/\eta_0$ y comparar con las notas.
\end{enumerate}
\end{ejercicio}

\begin{solucion}
        \begin{enumerate}
        \item  Si multiplicamos $x_c$ por el generador local $2\cos(\omega_c t + \pi/4)$ y despreciamos los términos de alta frecuencia, nos queda:
        \begin{align*}
                x_1(t) &= \sum_k Ap_R(t-kT_s)\cos(\omega_c t + \theta_k)2\cos(\omega_c t + \pi/4) \\
                &= \sum_k Ap_R(t-kT_s)\cos(\theta_k-\pi/4) + \ldots
        \end{align*}
        Con el mismo razonamiento, multiplicando por el otro oscilador:
        \begin{align*}
                x_2(t) &= \sum_k Ap_R(t-kT_s)\cos(\omega_c t + \theta_k)2\cos(\omega_c t + 3\pi/4) \\
                &= \sum_k Ap_R(t-kT_s)\cos(\theta_k-3\pi/4) + \ldots
        \end{align*}

        Notemos que si $\theta_k=0$ o $\pi/2$, $x_1$ produce la misma señal de amplitud $(A/\sqrt{2})p_R$, pero $x_2$ cambia de signo. Razonando de este modo, hacemos un filtro acoplado a cada una de las señales y regeneramos de acuerdo a:
        \begin{equation*}
                \begin{cases}
                        x_1 = A/\sqrt{2}, x_2 = -A/\sqrt{2} & \mapsto \theta_k = 0,\\
                        x_1 = A/\sqrt{2}, x_2 = A/\sqrt{2} & \mapsto \theta_k = \pi/2,\\
                        x_1 = -A/\sqrt{2}, x_2 = A/\sqrt{2} & \mapsto \theta_k = \pi,\\
                        x_1 = -A/\sqrt{2}, x_2 = -A/\sqrt{2} & \mapsto \theta_k = 3\pi/2.
                \end{cases}
        \end{equation*}

        \item Para el sistema anterior, la probabilidad de error de símbolo es igual a un sistema $4-QAM$ con amplitud $A/\sqrt{2}$. En cada señal usamos $0$ como umbral por lo que la probabilidad de error en $x_1$ o en $x_2$ es $Q(A/(\sqrt{2}\sigma))$ y la probabilidad total es $P_e\approx 2Q(A/(\sqrt{2}\sigma))$.
        
        Observando que $\bar{E}_p = A^2 T_s/2$ (pulso de RF de amplitud $A$) y $\sigma^2=\eta_0 f_s$, se tiene $\frac{A^2}{2\sigma^2}=\frac{A^2T_s}{2\eta_0}=\frac{\bar E_p}{\eta_0}$, y la expresión queda $P_e \approx 2Q\left(\sqrt{\frac{\bar{E}_p}{\eta_0}}\right) = 2Q(\sqrt{\gamma_s})$. Como cada símbolo lleva $2$ bits, $\gamma_s=2\gamma$ y $P_e\approx 2Q(\sqrt{2\gamma})$, que es el resultado de las notas para QPSK: el detector con osciladores a $\pm\pi/4$ es simplemente el detector coherente de $4$-QAM aplicado a la constelación rotada.
\end{enumerate}
\end{solucion}

\begin{ejercicio}
En los sistemas coherentes donde se debe enviar una pequeña señal piloto para la sincronización, empeora la eficiencia. Considere un sistema digital BPSK, cuya onda trasmitida es
\begin{equation*}
        x_c(t) = \sum_k a_k\sqrt{1-\mu^2}p_R(t-kT_s)\cos(\omega_c t)+\mu
        A\sin(\omega_c t),
\end{equation*}
donde $a_k=\{+A,-A\}$, $0<\mu<1$, y $p_R(t)$ es el pulso rectangular NRZ de largo $T_s$. Calcule la probabilidad de error de símbolo en función de la densidad espectral de potencia del ruido $\frac{\eta_0}{2}$ y la potencia media trasmitida $S$. Compare el resultado con el que se obtendría si no se enviara la portadora piloto $s_c(t)=\mu A\sin(\omega_c t)$.
\end{ejercicio}

\begin{solucion}
La onda enviada tiene una componente BPSK y una componente de portadora. La componente BPSK tiene amplitudes $a'_k = \pm A\sqrt{1-\mu^2}$ por lo que la potencia es $S_{\mathrm{bpsk}} = E[(a'_k)^2] f_s ||\tilde{p}||^2$ siendo $||\tilde{p}||$ el pulso NRZ en radiofrecuencia, de energía $T_s/2$. Por lo tanto: 
\begin{equation*}
        S_{\mathrm{bpsk}}=\frac{1}{2}E[(a'_k)^2] = \frac{A^2 (1-\mu^2)}{2}.        
\end{equation*}

La parte sinusoidal de portadora es de potencia $\frac{A^2\mu^2}{2}$ por lo que la potencia total transmitida es $S=\frac{A^2}{2}$ (las dos componentes están en cuadratura, por lo que sus potencias se suman). Es decir, usamos la misma potencia total que un BPSK común de amplitud $A$, pero reservamos una fracción $\mu^2$ de esa potencia para la portadora.

Si aplicamos detección coherente a la parte BPSK (la piloto, en cuadratura, no aparece en la rama en fase), tenemos que usar $A' = A\sqrt{1-\mu^2}$, por lo que sustituyendo en la expresión de BPSK queda:
\begin{equation*}
        P_e = Q\left(A'/\sigma\right) = Q\left(\sqrt{\frac{A^2(1-\mu^2)}{\eta_0 f_s}}\right) = Q\left(\sqrt{\frac{2S(1-\mu^2)}{\eta_0 f_s}}\right)= Q\left(\sqrt{2\gamma(1-\mu^2)}\right),
\end{equation*}
donde $\gamma = \bar{E}_b/\eta_0$, con $\bar{E}_b = ST_s=A^2T_s/2$ la energía total transmitida por bit (incluyendo la piloto). Sin piloto ($\mu=0$) se recupera $Q(\sqrt{2\gamma})$: la piloto cuesta $-10\log_{10}(1-\mu^2)\ \unit{\decibel}$. A continuación una comparación del efecto de $\mu$ para $\gamma = 10$ dB.

\end{center}

\end{solucion}

\item Relacionar este detector con el problema de la ambigüedad de fase del capítulo de sincronización: ¿qué ocurre con $m_k$ si la portadora llega con una fase constante desconocida $\theta$?
\end{enumerate}

\end{ejercicio}

\begin{solucion}

        \begin{enumerate}
                \item La señal a la entrada del integrador (ignorando el ruido), denotando $\theta_k=a_k\pi$ es:
                \begin{align*}
                        y(t) &= 2x_c(t)x_c(t-T_s) \\
                        &= 2\left(\sum_k Ap(t-kT_s)\cos(\omega_c t + \theta_k)\right)\left(\sum_{k'} Ap(t-T_s-k'T_s)\cos(\omega_c t + \theta_k')\right)
                \end{align*}
                Notar que en el producto, los pulsos están desfasados y por lo tanto los productos cruzados dan $0$, excepto cuando $k'=k-1$, que da $p^2=p$ en el intervalo $[kT_s,(k+1)T_s)$, de donde:
                \begin{align*}
                        y(t) &= 2\sum_k A^2p(t-kT_s)\cos(\omega_c t + \theta_k)\cos(\omega_c (t-T_s) + \theta_{k-1}) \\
                             &= \sum_k A^2p(t-kT_s)\cos(\theta_k-\theta_{k-1}) + \ldots
                \end{align*}
                donde despreciamos los términos de alta frecuencia que se irán al filtrar (aquí usamos que en un tiempo de símbolo hay una cantidad entera de oscilaciones de la portadora, por hipótesis, por lo que $\omega_cT_s$ es múltiplo de $2\pi$).

                Integrando en un período hallamos $z_k$:
                \begin{equation*}
                        z_k = \frac{1}{T_s}\int_{kT_s}^{(k+1)T_s} y(t)dt = A^2\cos(\theta_k-\theta_{k-1}).
                \end{equation*}
                Como $\theta_k-\theta_{k-1}\in\{0,\pm\pi\}$, resulta $z_k = A^2$ si $\theta_k = \theta_{k-1}$ y $z_k=-A^2$ si $\theta_k\neq\theta_{k-1}$. Por eso se denomina demodulador diferencial: solo detecta cambios de símbolo, es decir diferencias de fase.
                Resumiendo:
                \begin{equation*}
                        \begin{cases}
                                m_k = 1 & \text{si }a_k = a_{k-1}\\
                                m_k = 0 & \text{si }a_k \neq a_{k-1}
                        \end{cases}
                        \qquad\text{es decir}\qquad m_k = 1\oplus a_k\oplus a_{k-1}.
                \end{equation*}

                \item Usando la idea anterior, partiendo de $a_0=1$, se mantiene el valor cuando $m_k=1$ y se invierte cuando $m_k=0$: $a_1\ldots a_8 = 11010001$.
                
                \item Tomando $\gamma = \bar{E}_p/\eta_0$, se tiene que:
                 \begin{align*}
                        P_e(\text{BPSK coh}) &= Q(\sqrt{2\gamma}), \\
                        P_e(\text{DBPSK} ) &= \frac{1}{2} e^{-\gamma}.
                 \end{align*}
                
                 Usando la aproximación $Q(x) \approx \frac{1}{\sqrt{2\pi} x}e^{-x^2/2}$, podemos escribir:
                 \begin{equation*}
                        \frac{P_e(\text{DBPSK})}{P_e(\text{BPSK coh})} = \frac{\frac{1}{2}e^{-\gamma}}{\frac{1}{\sqrt{4\pi\gamma}}e^{-\gamma}} = \sqrt{\pi\gamma}.
                 \end{equation*}
                 
                 La performance de DBPSK es levemente peor: el cociente crece solo como $\sqrt{\gamma}$, lo que en términos de energía equivale a una diferencia de menos de $1\ \unit{\decibel}$ para probabilidades de error bajas. Graficamos a continuación ambas en función de $\gamma$:
                 \begin{center}
                        \input{ej3_6.tikz}
                 \end{center}

                \item $\frac12\e^{-\gamma}=10^{-3}\Rightarrow \gamma=\ln 500\approx 6.21$ ($7.9\ \unit{\decibel}$). Con $\eta_0=2\times10^{-12}\ \unit{\square\volt/\hertz}$, $\bar E_p=6.21\,\eta_0\approx 1.24\times10^{-11}\ \unit{\square\volt\second}$, y la potencia es $S=\bar E_p f_s\approx 1.24\times10^{-6}\ \unit{\square\volt}$, es decir unos $-29\ \mathrm{dBm}$ sobre $1\ \unit{\ohm}$. (Con BPSK coherente harían falta $\gamma\approx 6.8\ \unit{\decibel}$, $1.1\ \unit{\decibel}$ menos.)

                \item Si la portadora llega con fase $\theta$ desconocida, todos los $\theta_k$ se corren en $\theta$, y $z_k=A^2\cos(\theta_k-\theta_{k-1})$ no cambia: el detector no necesita recuperar la fase de la portadora. Es la misma idea que la codificación diferencial del capítulo de sincronización, que resuelve la ambigüedad de fase de $\pi$ del lazo de Costas, aquí llevada al extremo de no usar oscilador local. El precio es usar como referencia el símbolo anterior, que también tiene ruido.
        \end{enumerate}
        
\end{solucion}

\begin{solucion}
\begin{enumerate}
\item $Z_0=\sqrt{L/C}=\sqrt{294\times10^{-9}/52.3\times10^{-12}}\approx75\ \unit{\ohm}$ y $v_p=1/\sqrt{LC}\approx2.55\times10^8\ \unit{\meter/\second}$, es decir $0.85c$ (dieléctrico de espuma).
\item[2. y 3.] Calculando $\gamma=\sqrt{(R+j2\pi fL)(j2\pi fC)}$ y $Z_0=(R+j2\pi fL)/\gamma$:
\begin{center}
\begin{tabular}{c|c|c|c|c|c}
$f$ & $2\pi fL$ ($\unit{\ohm/\meter}$) & $\gamma$ ($\unit{\per\meter}$) & $Z_0$ ($\unit{\ohm}$) & $\alpha$ ($\unit{\decibel}/100\ \unit{\meter}$) & $v_p$\\ \hline
$1\ \unit{\kilo\hertz}$ & $0.0018$ & $7.9\times10^{-5}+j8.3\times10^{-5}$ & $252-j241$ & $0.07$ & $0.25c$\\
$100\ \unit{\kilo\hertz}$ & $0.18$ & $2.65\times10^{-4}+j2.48\times10^{-3}$ & $75.4-j8.1$ & $0.23$ & $0.85c$\\
$10\ \unit{\mega\hertz}$ & $18.5$ & $2.67\times10^{-4}+j0.246$ & $75.0-j0.1$ & $0.23$ & $0.85c$
\end{tabular}
\end{center}
La aproximación de bajas pérdidas exige $R\ll2\pi fL$, es decir $f\gg R/(2\pi L)\approx22\ \unit{\kilo\hertz}$. A $100\ \unit{\kilo\hertz}$ ya vale razonablemente, y a $10\ \unit{\mega\hertz}$ muy bien: en ambos casos $\alpha\approx R/(2Z_0)=2.67\times10^{-4}\ \unit{\neper/\meter}$, es decir $0.23\ \unit{\decibel}$ cada $100\ \unit{\meter}$, independiente de la frecuencia. A $1\ \unit{\kilo\hertz}$ no vale: $Z_0$ es fuertemente complejo y la velocidad de fase cae a menos de un tercio. A bajas frecuencias la línea se comporta como una red $RC$ distribuida, con $\alpha\approx\beta\approx\sqrt{\pi fRC}$, y es dispersiva.
\item[4.] Con $R$ y $G$ constantes, en el régimen de bajas pérdidas $\alpha$ es constante y $\beta=2\pi f/v_p$ es lineal en $f$: la línea solo atenúa y retarda, sin distorsionar. La ISI de un cable real proviene del efecto pelicular: $R$ crece como $\sqrt f$, lo que hace crecer $\alpha$ con la frecuencia y agrega una fase no lineal.
\end{enumerate}
\end{solucion}


\begin{ejercicio}[Bobinas de carga en un par telefónico]
Un par telefónico tiene $R=170\ \unit{\ohm/\kilo\meter}$, $L=0.6\ \unit{\milli\henry/\kilo\meter}$, $C=50\ \unit{\nano\farad/\kilo\meter}$ y $G\approx 0$.
\begin{enumerate}
\item Verificar que en la banda de voz ($300$ a $3000\ \unit{\hertz}$) el par \emph{no} está en el régimen de bajas pérdidas. Mostrar que en ese caso $\gamma(f)\approx\sqrt{j2\pi f RC}$, y por lo tanto $\alpha(f)\approx\sqrt{\pi f RC}$. Calcular $\alpha$ en $\unit{\decibel/\kilo\meter}$ a $300$, $1000$ y $3000\ \unit{\hertz}$, y comentar la distorsión que produce.
\item Se intercalan bobinas de $88\ \unit{\milli\henry}$ cada $1830\ \unit{\meter}$. Calcular la inductancia efectiva por unidad de longitud y determinar a partir de qué frecuencia vale ahora la aproximación de bajas pérdidas. Calcular $Z_0$ y $\alpha$ en la banda de voz (con la aproximación y en forma exacta) y comparar con la parte anterior.
\item La cadena de bobinas y la capacidad de cada tramo forman una escalera $LC$ pasabajos, con frecuencia de corte $f_{\max}\approx 1/(\pi\sqrt{L_b C_t})$, siendo $L_b$ la inductancia de cada bobina y $C_t$ la capacidad de cada tramo. Estimar $f_{\max}$ y explicar por qué fue necesario retirar las bobinas para ofrecer ADSL.
\end{enumerate}
\end{ejercicio}
\begin{solucion}
\begin{enumerate}
\item $2\pi fL=1.1$, $3.8$ y $11.3\ \unit{\ohm/\kilo\meter}$ a $300$, $1000$ y $3000\ \unit{\hertz}$, mucho menores que $R=170\ \unit{\ohm/\kilo\meter}$. Despreciando $j2\pi fL$ frente a $R$, $\gamma\approx\sqrt{j2\pi fRC}=\sqrt{2\pi fRC}\,\e^{j\pi/4}$, de donde $\alpha=\beta=\sqrt{\pi fRC}$. Esto da $0.78$, $1.42$ y $2.46\ \unit{\decibel/\kilo\meter}$ (el cálculo exacto da $0.77$, $1.40$ y $2.38$). La atenuación se triplica dentro de la banda de voz, y además $v_p=2\pi f/\beta\propto\sqrt f$: el par sin carga distorsiona tanto en amplitud como en fase.
\item $L_{\text{ef}}=0.6+88/1.83\approx48.7\ \unit{\milli\henry/\kilo\meter}$. Ahora $R=2\pi fL_{\text{ef}}$ para $f\approx560\ \unit{\hertz}$, y $2\pi fL_{\text{ef}}=306$ y $918\ \unit{\ohm/\kilo\meter}$ a $1$ y $3\ \unit{\kilo\hertz}$: la aproximación de bajas pérdidas vale razonablemente a partir de $1\ \unit{\kilo\hertz}$ (y no a $300\ \unit{\hertz}$). Allí $Z_0\approx\sqrt{L_{\text{ef}}/C}\approx990\ \unit{\ohm}$ y $\alpha\approx R/(2Z_0)\approx0.75\ \unit{\decibel/\kilo\meter}$. El cálculo exacto da $\alpha=0.60$, $0.72$ y $0.75\ \unit{\decibel/\kilo\meter}$ y $|Z_0|=1430$, $1055$ y $995\ \unit{\ohm}$. La atenuación a $1\ \unit{\kilo\hertz}$ se reduce a la mitad (el alcance para el mismo presupuesto en $\unit{\decibel}$ se duplica) y, sobre todo, varía poco dentro de la banda de voz.
\item $C_t=50\ \unit{\nano\farad/\kilo\meter}\times1.83\ \unit{\kilo\meter}\approx91.5\ \unit{\nano\farad}$, y
\begin{equation*}
f_{\max}\approx\frac{1}{\pi\sqrt{0.088\times91.5\times10^{-9}}}\approx3.5\ \unit{\kilo\hertz}.
\end{equation*}
Por encima de esa frecuencia la escalera prácticamente no deja pasar señal, y ADSL usa frecuencias de hasta $1.1\ \unit{\mega\hertz}$.
\end{enumerate}
\end{solucion}


\begin{ejercicio}[Reflexión, ROE y potencia entregada]
Una línea sin pérdidas de $Z_0=50\ \unit{\ohm}$ es alimentada por un generador sinusoidal adaptado ($Z_G=Z_0$).
\begin{enumerate}
\item Para las cargas $Z_L=75\ \unit{\ohm}$, $10\ \unit{\ohm}$, $30-j40\ \unit{\ohm}$ y $j50\ \unit{\ohm}$, calcular $\Gamma_L$, la ROE, la \emph{pérdida de retorno} $-20\log_{10}|\Gamma_L|$ y la fracción de la potencia incidente que se entrega a la carga. Ubicar las cuatro cargas en la Carta de Smith.
\item Para cada carga, hallar la distancia desde la carga al primer mínimo de la onda estacionaria de tensión.
\item Mostrar que, con el generador adaptado, la potencia entregada a la carga es $(1-|\Gamma_L|^2)$ veces la potencia disponible del generador.
\item Sobre una línea de $50\ \unit{\ohm}$ con una carga desconocida se mide una ROE de $3$, y el primer mínimo de tensión a $\lambda/8$ de la carga. Hallar $Z_L$, analíticamente y usando la Carta de Smith.
\item Para la carga de $10\ \unit{\ohm}$, diseñar un transformador de cuarto de onda. ¿Por qué no puede usarse directamente para la carga $30-j40\ \unit{\ohm}$? ¿Cómo podría usarse igual, intercalando antes un tramo de línea?
\item En la simulación de un pulso enviado por una línea \emph{con pérdidas}, con generador adaptado, la carga recibe el $67.7\%$ de la energía disponible cuando $Z_L=Z_0$, y el $44.6\%$ cuando $Z_L=4Z_0$. Explicar ambos números: ¿dónde está el $32\%$ que falta en el primer caso? ¿Por qué el cociente entre ambos es aproximadamente $0.64$?
\end{enumerate}
\end{ejercicio}
\begin{solucion}
\begin{enumerate}
\item Con $\zeta_L=Z_L/Z_0$ y $\Gamma_L=\frac{\zeta_L-1}{\zeta_L+1}$:
\begin{center}
\begin{tabular}{c|c|c|c|c}
$Z_L$ ($\unit{\ohm}$) & $\Gamma_L$ & ROE & pérdida de retorno & $1-|\Gamma_L|^2$\\ \hline
$75$ & $0.2$ & $1.5$ & $14\ \unit{\decibel}$ & $96\%$\\
$10$ & $-2/3$ & $5$ & $3.5\ \unit{\decibel}$ & $56\%$\\
$30-j40$ & $-j0.5$ & $3$ & $6\ \unit{\decibel}$ & $75\%$\\
$j50$ & $j$ & $\infty$ & $0\ \unit{\decibel}$ & $0\%$
\end{tabular}
\end{center}
En la carta, las dos cargas resistivas están sobre el eje real ($75\ \unit{\ohm}$ a la derecha del centro, $10\ \unit{\ohm}$ a la izquierda), $30-j40$ en el semiplano inferior (capacitivo), sobre el eje imaginario, y $j50$ sobre la circunferencia exterior, en el punto $\Gamma=j$ (reactancia pura).
\item A distancia $d$ de la carga, $\Gamma(d)=\Gamma_L\e^{-j4\pi d/\lambda}$, y el mínimo de tensión está donde $\Gamma(d)=-|\Gamma_L|$. Si $\Gamma_L=|\Gamma_L|\e^{j\theta_L}$, se necesita $\theta_L-4\pi d/\lambda=-\pi$ (módulo $2\pi$), es decir $d_{\min}=\frac{\theta_L+\pi}{4\pi}\lambda$, módulo $\lambda/2$. Resulta $\lambda/4$ para $75\ \unit{\ohm}$ ($\theta_L=0$), $0$ para $10\ \unit{\ohm}$ (el mínimo está en la carga), $\lambda/8$ para $30-j40\ \unit{\ohm}$ ($\theta_L=-\pi/2$) y $3\lambda/8$ para $j50\ \unit{\ohm}$ ($\theta_L=\pi/2$). En la carta, es lo que hay que girar en sentido horario hasta el semieje real negativo.
\item Con $Z_G=Z_0$, la onda incidente tiene amplitud $V^+=V_s\frac{Z_0}{Z_0+Z_G}=\frac{V_s}{2}$ y potencia $\frac{|V^+|^2}{2Z_0}=\frac{|V_s|^2}{8Z_0}$, que es la potencia disponible de un generador de impedancia interna $Z_0$. La onda reflejada lleva una fracción $|\Gamma_L|^2$ de esa potencia, y el generador adaptado la absorbe sin volver a reflejarla, por lo que la carga recibe $(1-|\Gamma_L|^2)P_{\text{disp}}$.
\item $|\Gamma_L|=\frac{\mathrm{ROE}-1}{\mathrm{ROE}+1}=0.5$. En el mínimo, $\Gamma(\lambda/8)=\Gamma_L\e^{-j\pi/2}=-0.5$, de donde $\Gamma_L=-0.5\,\e^{j\pi/2}=-j0.5$ y
\begin{equation*}
Z_L=Z_0\frac{1+\Gamma_L}{1-\Gamma_L}=50\,\frac{1-j0.5}{1+j0.5}=30-j40\ \unit{\ohm},
\end{equation*}
que es una de las cargas de la parte 1. En la carta: se traza la circunferencia de ROE $3$, se ubica el mínimo en el semieje real negativo ($\zeta=1/3$), y se gira $\lambda/8$ en sentido antihorario (hacia la carga) hasta $\zeta_L=0.6-j0.8$.
\item $Z_0'=\sqrt{50\times10}\approx22.4\ \unit{\ohm}$, en un tramo de longitud $\lambda/4$. Para una carga compleja, el tramo de $\lambda/4$ da $Z_0'^2/Z_L$, que también es complejo, y no puede igualar a $50\ \unit{\ohm}$ con ningún $Z_0'$ real. Pero a distancia $d_{\min}=\lambda/8$ la impedancia vista es real, $Z_0/\mathrm{ROE}\approx16.7\ \unit{\ohm}$: intercalando primero $\lambda/8$ de la línea de $50\ \unit{\ohm}$ y luego un transformador de $\lambda/4$ con $Z_0'=\sqrt{50\times16.7}\approx28.9\ \unit{\ohm}$, la línea queda adaptada.
\item Con $Z_L=Z_0$ no hay reflexión: el $32\%$ que falta se disipó en la línea, que atenúa el pulso. Con $Z_L=4Z_0$, $\Gamma_L=0.6$, y de la energía que llega la carga toma $1-0.36=0.64$: $0.677\times0.64\approx0.433$, cerca del $44.6\%$ simulado. La pequeña diferencia se debe a que en la línea con pérdidas $Z_0(f)$ es complejo y depende de la frecuencia, por lo que $\Gamma_L(f)$ no es exactamente $0.6$ en toda la banda del pulso.
\end{enumerate}
\end{solucion}


\begin{ejercicio}[Adaptación con stub]
Sobre una línea sin pérdidas de $Z_0=50\ \unit{\ohm}$ se conecta una carga $Z_L=20-j30\ \unit{\ohm}$.
\begin{enumerate}
\item Calcular $\Gamma_L$ y la ROE. Ubicar la carga en la Carta de Smith.
\item Usando la carta como carta de admitancias, hallar las dos distancias $d$ a la carga en las que puede conectarse un stub cortocircuitado en paralelo, y la susceptancia que debe aportar en cada caso.
\item Hallar las longitudes $\ell_s$ de los stubs correspondientes, y verificar analíticamente ambas soluciones.
\item ¿Cuál de las dos soluciones es preferible, y por qué?
\end{enumerate}
\end{ejercicio}
\begin{solucion}
\begin{enumerate}
\item $\zeta_L=0.4-j0.6$, $\Gamma_L=\frac{\zeta_L-1}{\zeta_L+1}\approx-0.207-j0.517$, $|\Gamma_L|\approx0.557$ y $\mathrm{ROE}\approx3.52$. En la carta, la carga está en el semiplano inferior (capacitiva), sobre el círculo de ROE $3.52$.
\item La admitancia es $y_L=1/\zeta_L\approx0.77+j1.15$, el punto diametralmente opuesto en la carta. Girando en sentido horario (hacia el generador) sobre el círculo de ROE se corta la circunferencia $g=1$ en dos puntos: $y=1+j1.34$ a $d_1\approx0.017\lambda$ y $y=1-j1.34$ a $d_2\approx0.173\lambda$. El stub debe aportar $-j1.34$ en el primer caso y $+j1.34$ en el segundo.
\item Con $y_{cc}=-j\cot(\beta\ell_s)$: para $-j1.34$, $\cot(\beta\ell_s)=1.34$, $\beta\ell_s\approx0.640\ \unit{\radian}$ y $\ell_s\approx0.102\lambda$; para $+j1.34$, $\beta\ell_s\approx\pi-0.640$ y $\ell_s\approx0.398\lambda$. Verificación: con
\begin{equation*}
y(d)=\frac{y_L+j\tan(\beta d)}{1+jy_L\tan(\beta d)}
\end{equation*}
se obtiene $y(0.0167\lambda)\approx1.00+j1.34$ y $y(0.1727\lambda)\approx1.00-j1.34$.
\item La primera: el stub y la distancia a la carga son más cortos, por lo que la adaptación es menos sensible a la frecuencia (cuanto más largos los tramos, más rápido cambian sus fases con $f$), y la onda estacionaria queda confinada a un tramo más corto. En la práctica puede ser difícil conectar el stub a solo $0.017\lambda$ de la carga, y entonces se usa la segunda.
\end{enumerate}
\end{solucion}


\begin{ejercicio}[Diseño de una adaptación con stub y ancho de banda]
Una carga se modela como una resistencia de $100\ \unit{\ohm}$ en paralelo con una capacidad de $20\ \unit{\pico\farad}$. Se la quiere conectar, a $f_0=100\ \unit{\mega\hertz}$, a un cable coaxil de $Z_0=50\ \unit{\ohm}$ con velocidad de propagación $v_p=2\times10^8\ \unit{\meter/\second}$, sin pérdidas.
\begin{enumerate}
\item Calcular la admitancia normalizada de la carga a $f_0$, $\Gamma_L$ y la ROE sin adaptar.
\item Hallar las dos soluciones de adaptación con un stub en paralelo, y dar la distancia $d$ y la longitud del stub \emph{en centímetros}, tanto para un stub cortocircuitado como para uno en circuito abierto. ¿Cuál es la combinación más corta?
\item Para la primera solución con stub cortocircuitado, calcular la ROE vista desde el cable a $90$, $95$, $105$ y $110\ \unit{\mega\hertz}$, teniendo en cuenta que cambian tanto las longitudes eléctricas $\beta d$ y $\beta\ell_s$ como la admitancia de la carga. Estimar la banda de frecuencias en la que la ROE se mantiene por debajo de $1.5$.
\end{enumerate}
\end{ejercicio}
\begin{solucion}
\begin{enumerate}
\item A $f_0$, $Y_L=\frac{1}{100}+j2\pi f_0\,20\times10^{-12}=0.01+j0.0126\ \unit{\siemens}$, y $y_L=Z_0Y_L=0.5+j0.628$. Entonces $Z_L=1/Y_L\approx38.8-j48.7\ \unit{\ohm}$, $\Gamma_L\approx0.134-j0.475$, $|\Gamma_L|\approx0.494$ y $\mathrm{ROE}\approx2.95$.
\item $\lambda=v_p/f_0=2\ \unit{\meter}$. Girando $y_L$ hacia el generador sobre el círculo de ROE $2.95$:
\begin{itemize}
\item $d_1\approx0.063\lambda=12.6\ \unit{\centi\meter}$, con $y(d_1)=1+j1.14$. El stub debe aportar $-j1.14$: cortocircuitado, $\cot(\beta\ell_s)=1.14$, $\ell_s\approx0.115\lambda=23.0\ \unit{\centi\meter}$; en circuito abierto, $y_{ca}=j\tan(\beta\ell_s)=-j1.14$, $\ell_s\approx0.365\lambda=73.0\ \unit{\centi\meter}$.
\item $d_2\approx0.231\lambda=46.2\ \unit{\centi\meter}$, con $y(d_2)=1-j1.14$. El stub debe aportar $+j1.14$: cortocircuitado, $\ell_s\approx0.385\lambda=77.0\ \unit{\centi\meter}$; en circuito abierto, $\ell_s\approx0.135\lambda=27.0\ \unit{\centi\meter}$.
\end{itemize}
La combinación más corta es la primera con stub cortocircuitado: $12.6+23.0\ \unit{\centi\meter}$. Para cada distancia, uno de los dos tipos de stub es corto ($<\lambda/4$) y el otro largo: los dos difieren en $\lambda/4$.
\item Con $d=12.6\ \unit{\centi\meter}$ y $\ell_s=23.0\ \unit{\centi\meter}$ fijos, a cada frecuencia se recalcula $\beta=2\pi f/v_p$, la admitancia de la carga $Y_L(f)$, la admitancia $y(d)$ vista en el punto de conexión, y se le suma $y_{cc}(\ell_s)=-j\cot(\beta\ell_s)$. La ROE vista desde el cable resulta:
\begin{center}
\begin{tabular}{c|ccccc}
$f$ ($\unit{\mega\hertz}$) & $90$ & $95$ & $100$ & $105$ & $110$\\ \hline
ROE & $1.47$ & $1.20$ & $1.00$ & $1.19$ & $1.40$
\end{tabular}
\end{center}
La ROE se mantiene por debajo de $1.5$ aproximadamente entre $89.5$ y $112\ \unit{\mega\hertz}$, una banda de unos $23\ \unit{\mega\hertz}$ ($23\%$ de $f_0$). La adaptación con stub es exacta en una sola frecuencia y se degrada al alejarse de ella.
\end{enumerate}
\end{solucion}


\begin{ejercicio}[Efecto pelicular y presupuesto de pérdidas]
Un par trenzado de conductores de cobre AWG24 (radio $a=0.26\ \unit{\milli\meter}$) tiene una atenuación de $3.81\ \unit{\decibel}$ cada $100\ \unit{\meter}$ a $10\ \unit{\mega\hertz}$, dominada por el efecto pelicular. Para el cobre, $\sigma=5.96\times10^7\ \unit{\siemens/\meter}$ y $\mu=4\pi\times10^{-7}\ \unit{\henry/\meter}$.
\begin{enumerate}
\item Calcular la profundidad de penetración en $50\ \unit{\hertz}$, $1\ \unit{\mega\hertz}$ y $100\ \unit{\mega\hertz}$. ¿A qué frecuencia es igual al radio del conductor? ¿Qué significa que esté por debajo o por encima de esa frecuencia?
\item Calcular la atenuación cada $100\ \unit{\meter}$ a $2.5$, $31.25$ y $100\ \unit{\mega\hertz}$.
\item Si cada tramo tolera una atenuación de $20\ \unit{\decibel}$, evaluada a la velocidad de señalización, calcular la separación entre repetidores para $f_s=10\ \unit{\mega\baud}$ y para $f_s=40\ \unit{\mega\baud}$. ¿Qué habría pasado si la atenuación creciera proporcionalmente a $f$?
\item A partir de la respuesta en frecuencia del cable de las notas, calcular el producto $k\ell$ para $100\ \unit{\meter}$ y el retardo de grupo adicional $\tau_g(f)-t_d$ a $1$ y $10\ \unit{\mega\hertz}$. Repetir para $1\ \unit{\kilo\meter}$ y comparar la diferencia con el tiempo de símbolo para $f_s=10\ \unit{\mega\baud}$.
\end{enumerate}
\end{ejercicio}
\begin{solucion}
\begin{enumerate}
\item $\delta=1/\sqrt{\pi f\mu\sigma}$ da $9.2\ \unit{\milli\meter}$ a $50\ \unit{\hertz}$, $65\ \unit{\micro\meter}$ a $1\ \unit{\mega\hertz}$ y $6.5\ \unit{\micro\meter}$ a $100\ \unit{\mega\hertz}$. Es igual al radio, $\delta=a$, para $f=1/(\pi\mu\sigma a^2)\approx63\ \unit{\kilo\hertz}$. Por debajo de esa frecuencia la corriente ocupa toda la sección del conductor y la resistencia es la de continua. Por encima, la corriente se concentra en una corona de espesor $\delta$, la sección efectiva disminuye como $1/\sqrt f$ y $R$ crece como $\sqrt f$. Por eso la resistencia de un cable de energía a $50\ \unit{\hertz}$ no depende de la frecuencia, y la de un cable de datos sí.
\item $\alpha(f)=3.81\sqrt{f/10\ \unit{\mega\hertz}}$: $1.9$, $6.7$ y $12.0\ \unit{\decibel}$ cada $100\ \unit{\meter}$.
\item $\ell=20/3.81\times100\approx525\ \unit{\meter}$ a $10\ \unit{\mega\baud}$, y $262\ \unit{\meter}$ a $40\ \unit{\mega\baud}$: cuadruplicar la velocidad reduce la distancia a la mitad. Si la atenuación fuera proporcional a $f$, se reduciría a la cuarta parte, $131\ \unit{\meter}$.
\item $\alpha(f)\ell=k\ell\sqrt f$ en neper: $k\ell=\frac{3.81/8.686}{\sqrt{10^7}}\approx1.39\times10^{-4}\ \unit{\second^{1/2}}$ para $100\ \unit{\meter}$. Entonces
\begin{equation*}
\tau_g(f)-t_d=\frac{k\ell}{4\pi\sqrt f}\approx11.0\ \unit{\nano\second}\ \text{a}\ 1\ \unit{\mega\hertz},\qquad 3.5\ \unit{\nano\second}\ \text{a}\ 10\ \unit{\mega\hertz}.
\end{equation*}
Para $1\ \unit{\kilo\meter}$ los valores se multiplican por $10$: $110$ y $35\ \unit{\nano\second}$. La diferencia, unos $75\ \unit{\nano\second}$, es comparable al tiempo de símbolo de $100\ \unit{\nano\second}$, por lo que la cola del pulso se extiende sobre los símbolos siguientes y hace falta ecualizar.
\end{enumerate}
\end{solucion}


\begin{ejercicio}[Ecualizador de cero forzado]
Al medir la respuesta de un canal a un pulso aislado se obtienen las muestras $\tilde p_{-1}=0.1$, $\tilde p_0=1$, $\tilde p_1=0.3$, $\tilde p_2=0.1$, y $\tilde p_m\approx 0$ para el resto.
\begin{enumerate}
\item Calcular la ISI máxima (peor caso) para una señal binaria polar $\pm A$, en relación con $A$.
\item Diseñar un ecualizador de cero forzado de tres coeficientes. Calcular las muestras del pulso ecualizado $\hat p_k$ para $k=-3,\ldots,4$ y la nueva ISI máxima.
\item Si el ruido a la entrada del ecualizador está formado por muestras independientes de varianza $\sigma^2$, ¿cuál es su varianza a la salida? Expresar el aumento en $\unit{\decibel}$.
\end{enumerate}
\end{ejercicio}
\begin{solucion}
\begin{enumerate}
\item La ISI de peor caso es $A\sum_{m\neq0}|\tilde p_m|=0.5A$: el ojo queda abierto a la mitad.
\item El sistema de las notas, con $N=1$, es
\begin{equation*}
\begin{pmatrix}1&0.1&0\\0.3&1&0.1\\0.1&0.3&1\end{pmatrix}\begin{pmatrix}c_{-1}\\c_0\\c_1\end{pmatrix}=\begin{pmatrix}0\\1\\0\end{pmatrix},
\end{equation*}
de solución $c_{-1}\approx-0.106$, $c_0\approx1.063$, $c_1\approx-0.308$. El pulso ecualizado $\hat p_k=\sum_nc_n\tilde p_{k-n}$ vale
